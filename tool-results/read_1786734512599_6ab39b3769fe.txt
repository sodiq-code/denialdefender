   355→function mockReviewer(inputData: Record<string, unknown>) {
   356→  const draft = (inputData.draft ?? {}) as Record<string, unknown>;
   357→  const evidence = (inputData.evidence ?? {}) as Record<string, unknown>;
   358→  const citationsCount = Array.isArray(draft.citations_used) ? draft.citations_used.length : 3;
   359→  const evidenceStrength = (evidence.overall_evidence_strength as string) ?? "strong";
   360→
   361→  const checks = [
   362→    { category: "COMPLETENESS", status: "pass", score: 0.95, details: "All 9 required sections are present and contain substantive content.", severity: "info" },
   363→    { category: "CITATION_ACCURACY", status: "pass", score: 0.90, details: `${citationsCount} citations found with provenance tiers. All evidence items properly referenced.`, severity: "info" },
   364→    { category: "CLINICAL_ACCURACY", status: "pass", score: 0.88, details: `Clinical rationale aligns with evidence strength (${evidenceStrength}). Medical arguments are sound.`, severity: "info" },
   365→    { category: "TONE_APPROPRIATENESS", status: "pass", score: 0.92, details: "Tone is professional and factual. No adversarial or emotional language detected.", severity: "info" },
   366→    { category: "COMPLIANCE", status: "pass", score: 0.95, details: "No PHI detected. Patient identified by hash only. No SSN, DOB, or real names present.", severity: "info" },
   367→    { category: "PERSUASIVENESS", status: "pass", score: 0.85, details: "Arguments are logical and well-structured. Evidence hierarchy supports the case effectively.", severity: "info" },
   368→    { category: "FORMATTING", status: "pass", score: 0.90, details: "Business letter format followed. Clear section delineation and professional layout.", severity: "info" },
   369→    { category: "SPECIFICITY", status: "needs_improvement", score: 0.75, details: "Denial codes are referenced but payer-specific policy section numbers could be more precise.", severity: "minor" },
   370→  ];
   371→
   372→  const avgScore = Math.round((checks.reduce((s, c) => s + c.score, 0) / checks.length) * 1000) / 1000;
   373→  const critical = checks.filter((c) => c.severity === "critical" && c.status === "fail");
   374→  const minor = checks.filter((c) => c.severity === "minor" && c.status !== "pass");
   375→
   376→  let verdict = "APPROVED";
   377→  let revisionInstructions: string | null = null;
   378→  if (critical.length > 0) {
   379→    verdict = "NEEDS_REVISION";
   380→    revisionInstructions = "Fix critical issues: " + critical.map((c) => c.details).join("; ");
   381→  } else if (minor.length >= 2) {
   382→    verdict = "NEEDS_REVISION";
   383→    revisionInstructions = "Address minor issues for higher quality: " + minor.map((c) => c.details).join("; ");
   384→  }
   385→
   386→  return {
   387→    overall_verdict: verdict,
   388→    overall_score: avgScore,
   389→    checks,
   390→    critical_issues: critical.map((c) => c.details),
   391→    minor_issues: minor.map((c) => c.details),
   392→    recommendations: [
   393→      "Add specific payer medical policy section references",
   394→      "Consider including a provider attestation statement",
   395→      "Strengthen the policy contradiction arguments with direct quotes",
   396→    ],
   397→    revision_instructions: revisionInstructions,
   398→  };
   399→}

   403→function mockCoder(inputData: Record<string, unknown>) {
   404→  const denial = getDenial(inputData);
   405→  const cpt = denial.cpt_code ?? "99213";
   406→  const icd10 = denial.icd10_code ?? "M54.5";
   407→  const denialCode = denial.denial_code ?? "CO-50";
   408→
   409→  const issues: Array<{
   410→    category: string;
   411→    severity: string;
   412→    description: string;
   413→    original_code: string;
   414→    corrected_code: string | null;
   415→    correction_rationale: string;
   416→    would_reverse_denial: boolean;
   417→  }> = [];
   418→  let correctedCpt: string | null = null;
   419→  let correctedIcd10: string | null = null;
   420→
   421→  if (denialCode.startsWith("CO-4") || denialCode.startsWith("CO-11")) {
   422→    issues.push({
   423→      category: "CODE_DX_MATCH",
   424→      severity: "DIRECT_CAUSE",
   425→      description: `The ICD-10 code ${icd10} may not be a supported diagnosis for CPT ${cpt}. A more specific code may resolve the denial.`,
   426→      original_code: icd10,
   427→      corrected_code: "M54.16",
   428→      correction_rationale: "Increasing ICD-10 specificity to the appropriate subcategory",
   429→      would_reverse_denial: true,
   430→    });
   431→    correctedIcd10 = "M54.16";
   432→  }
   433→
   434→  if (denialCode.startsWith("CO-97")) {
   435→    issues.push({
   436→      category: "BUNDLING_ISSUES",
   437→      severity: "CONTRIBUTING",
   438→      description: `CPT ${cpt} may be bundled with another procedure. Adding modifier -59 may be appropriate.`,
   439→      original_code: cpt,
   440→      corrected_code: `${cpt}-59`,
   441→      correction_rationale: "Distinct procedural service modifier to unbundle appropriately",
   442→      would_reverse_denial: true,
   443→    });
   444→    correctedCpt = `${cpt}-59`;
   445→  }
   446→
   447→  let validationResult = "VALID";
   448→  if (issues.some((i) => i.severity === "DIRECT_CAUSE")) {
   449→    validationResult = "CORRECTABLE";
   450→  } else if (issues.length > 0) {
   451→    validationResult = "CORRECTABLE";
   452→  }
   453→
   454→  if (issues.length === 0) {
   455→    issues.push({
   456→      category: "CODE_SPECIFICITY",
   457→      severity: "UNRELATED",
   458→      description: `ICD-10 code ${icd10} could be more specific but this is not the primary cause of denial.`,
   459→      original_code: icd10,
   460→      corrected_code: null,
   461→      correction_rationale: "Increased specificity may strengthen the appeal but won't reverse the denial alone",
   462→      would_reverse_denial: false,
   463→    });
   464→  }
   465→
   466→  return {
   467→    validation_result: validationResult,
   468→    overall_assessment: correctedCpt || correctedIcd10
   469→      ? `CPT ${cpt} with ICD-10 ${icd10}: Coding corrections available that may help the appeal.`
   470→      : `CPT ${cpt} with ICD-10 ${icd10}: Codes appear valid. Denial is likely based on medical necessity rather than coding.`,
   471→    issues_found: issues,
   472→    corrected_codes: {
   473→      cpt: correctedCpt,
   474→      icd10: correctedIcd10,
   475→      modifiers: correctedCpt?.includes("-59") ? ["59"] : [],
   476→    },
   477→    coding_action_required: correctedCpt !== null || correctedIcd10 !== null,
   478→    confidence: 0.87,
   479→  };

   775→    stdout: "pipe",
   776→    stderr: "pipe",
   777→  });
   778→
   779→  // Set timeout
   780→  const timeoutPromise = new Promise<never>((_, reject) => {
   781→    setTimeout(() => {
   782→      proc.kill();
   783→      reject(new Error("Python subprocess timed out after 60 seconds"));
   784→    }, SUBPROCESS_TIMEOUT_MS);
   785→  });
   786→
   787→  try {
   788→    const exitCode = await Promise.race([proc.exited, timeoutPromise]);
   789→
   790→    if (exitCode !== 0) {
   791→      const stderr = await new Response(proc.stderr).text();
   792→      console.error(`[workflow] Python subprocess exited with code ${exitCode}: ${stderr}`);
   793→      throw new Error(`Python subprocess failed with exit code ${exitCode}`);
   794→    }
   795→
   796→    const stdout = await new Response(proc.stdout).text();
   797→    return JSON.parse(stdout) as Record<string, unknown>;
   798→  } catch (err) {
   799→    console.error(`[workflow] Python subprocess error: ${err}`);
   800→    // Fall back to mock data
   801→    return mockWorkflow(inputData);
   802→  }
   803→}
