   149→// ─── Component ─────────────────────────────────────────────────────────────
   150→
   151→export function Day6PipelinePanel() {
   152→  const [denialText, setDenialText] = useState(SAMPLE_LETTERS[0].text);
   153→  const [payer, setPayer] = useState('Medicare');
   154→  const [selectedSample, setSelectedSample] = useState('0');
   155→  const [state, setState] = useState<PipelineState>({
   156→    phase: 'idle', caseId: null, pipelineStatus: '', traces: [], traceChecklist: [],
   157→    gate1: null, gate2: null, triage: null, policyResearch: null, evidenceAssembly: null,
   158→    letterDrafting: null, qualityReview: null, letterVersion: 0, latencyMs: 0,
   159→    cachedTriageResult: null, cachedAdvocateResult: null,
   160→  });
   161→  const [loading, setLoading] = useState(false);
   162→  const [error, setError] = useState<string | null>(null);
   163→  const [editedLetter, setEditedLetter] = useState('');
   164→  const [editReason, setEditReason] = useState('');
   165→  const [showVersionHistory, setShowVersionHistory] = useState(false);
   166→  const [gateTestResult, setGateTestResult] = useState<Record<string, unknown> | null>(null);
   167→  const [gateTestLoading, setGateTestLoading] = useState(false);
   168→
   169→  const traceEndRef = useRef<HTMLDivElement>(null);
   170→  useEffect(() => { traceEndRef.current?.scrollIntoView({ behavior: 'smooth' }); }, [state.traces.length]);
   171→
   172→  // ── Run Phase 1 (up to Gate 1) ───────────────────────────────────────
   173→  const runPhase1 = useCallback(async () => {
   174→    setLoading(true);
   175→    setError(null);
   176→    setState(prev => ({ ...prev, phase: 'running-phase1', traces: [], traceChecklist: [] }));
   177→
   178→    try {
   179→      const res = await fetch('/api/full-pipeline', {
   180→        method: 'POST',
   181→        headers: { 'Content-Type': 'application/json' },
   182→        body: JSON.stringify({ denialText, payer }),
   183→      });
   184→      const data = await res.json();
   185→
   186→      if (!res.ok) throw new Error(data.error || 'Pipeline failed');
   187→
   188→      setState(prev => ({
   189→        ...prev,
   190→        phase: 'awaiting-gate1',
   191→        caseId: data.caseId,
   192→        pipelineStatus: data.pipelineStatus,
   193→        traces: data.traces || [],
   194→        traceChecklist: data.traceChecklist || [],
   195→        gate1: data.gate1,
   196→        triage: data.triage,
   197→        latencyMs: data.latencyMs,
   198→        cachedTriageResult: data.triage,
   199→        cachedAdvocateResult: data.advocate,
   200→      }));
   201→    } catch (err: unknown) {
   202→      setError(err instanceof Error ? err.message : 'Unknown error');
   203→      setState(prev => ({ ...prev, phase: 'idle' }));
   204→    } finally {
   205→      setLoading(false);
   206→    }
   207→  }, [denialText, payer]);
   208→
   209→  // ── Resolve Gate 1 ───────────────────────────────────────────────────
   210→  const resolveGate1 = useCallback(async (status: 'approved' | 'rejected') => {
   211→    if (!state.caseId) return;
   212→    setLoading(true);
   213→    setError(null);
   214→
   215→    try {
   216→      const res = await fetch('/api/full-pipeline/resume', {
   217→        method: 'POST',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({
   220→          caseId: state.caseId,
   221→          gateStatus: status,
   222→          cachedTriageResult: state.cachedTriageResult,
   223→          cachedAdvocateResult: state.cachedAdvocateResult,
   224→        }),
   225→      });
   226→      const data = await res.json();
   227→
   228→      if (!res.ok) throw new Error(data.error || 'Resume failed');
   229→
   230→      const newPhase = data.pipelineStatus === 'awaiting_gate2' ? 'awaiting-gate2'
   231→        : data.pipelineStatus === 'gate1_rejected' ? 'gate1-rejected'
   232→        : data.pipelineStatus === 'quality_review_failed' ? 'quality-failed'
   233→        : 'idle';
   234→
   235→      setState(prev => ({
   236→        ...prev,
   237→        phase: newPhase,
   238→        pipelineStatus: data.pipelineStatus,
   239→        traces: [...prev.traces, ...(data.traces || [])],
   240→        traceChecklist: data.traceChecklist || [],
   241→        gate1: data.gate1,
   242→        gate2: data.gate2,
   243→        policyResearch: data.policyResearch,
   244→        evidenceAssembly: data.evidenceAssembly,
   245→        letterDrafting: data.letterDrafting,
   246→        qualityReview: data.qualityReview,
   247→        letterVersion: data.letterVersion || 0,
   248→        latencyMs: data.latencyMs,
   249→      }));
   250→
   251→      if (data.letterDrafting?.appealLetter) {
   252→        setEditedLetter(data.letterDrafting.appealLetter);
   253→      }
   254→    } catch (err: unknown) {
   255→      setError(err instanceof Error ? err.message : 'Unknown error');
   256→    } finally {
   257→      setLoading(false);
   258→    }
   259→  }, [state.caseId, state.cachedTriageResult, state.cachedAdvocateResult]);
   260→
   261→  // ── Resolve Gate 2 ───────────────────────────────────────────────────
   262→  const resolveGate2 = useCallback(async (resolution: 'approved' | 'rejected') => {
   263→    if (!state.caseId) return;
   264→    setLoading(true);
   265→    setError(null);
   266→
   267→    try {
   268→      const res = await fetch('/api/full-pipeline/gate2', {
   269→        method: 'POST',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({
   272→          caseId: state.caseId,
   273→          resolution,
   274→          editedLetter: editedLetter !== state.letterDrafting?.appealLetter ? editedLetter : undefined,
   275→          editReason: editReason || undefined,
   276→        }),
   277→      });
   278→      const data = await res.json();
   279→
   280→      if (!res.ok) throw new Error(data.error || 'Gate 2 resolution failed');
   281→
   282→      setState(prev => ({
   283→        ...prev,
   284→        phase: data.gate2Status === 'approved' ? 'completed' : 'awaiting-gate2',
   285→        gate2: prev.gate2 ? { ...prev.gate2, status: data.gate2Status } : null,
   286→        letterVersion: data.letterVersion || prev.letterVersion,
   287→      }));
   288→
   289→      if (data.gate2Status === 'approved') {
   290→        // Submit the appeal automatically after approval
   291→        await fetch('/api/full-pipeline/gate2', {
   292→          method: 'POST',
   293→          headers: { 'Content-Type': 'application/json' },
   294→          body: JSON.stringify({ caseId: state.caseId, action: 'submit' }),
   295→        });
   296→      }
   297→    } catch (err: unknown) {
   298→      setError(err instanceof Error ? err.message : 'Unknown error');
   299→    } finally {
   300→      setLoading(false);
   301→    }
   302→  }, [state.caseId, editedLetter, editReason, state.letterDrafting]);
   303→
   304→  // ── Run Gate Test ────────────────────────────────────────────────────
   305→  const runGateTest = useCallback(async () => {
   306→    setGateTestLoading(true);
   307→    setGateTestResult(null);
   308→    try {
   309→      const res = await fetch('/api/full-pipeline/gate-test', {
   310→        method: 'POST',
   311→        headers: { 'Content-Type': 'application/json' },
   312→        body: JSON.stringify({ denialText, payer }),
   313→      });
   314→      const data = await res.json();
   315→      setGateTestResult(data);
   316→    } catch (err: unknown) {
   317→      setGateTestResult({ error: err instanceof Error ? err.message : 'Gate test failed' });
   318→    } finally {
   319→      setGateTestLoading(false);
   320→    }
   321→  }, [denialText, payer]);
   322→
   323→  // ── Sample letter selection ──────────────────────────────────────────
   324→  const handleSampleSelect = useCallback((idx: string) => {
   325→    setSelectedSample(idx);
   326→    const sample = SAMPLE_LETTERS[parseInt(idx)];
   327→    if (sample) {
   328→      setDenialText(sample.text);
   329→      setPayer(sample.payer);
   330→    }
   331→  }, []);
   332→
   333→  // ── Derived state ────────────────────────────────────────────────────
   334→  const isRunning = loading;
   335→  const agentStepStatus = (agent: string): 'pending' | 'running' | 'completed' | 'error' => {
   336→    const hasCompleted = state.traces.some(t => t.agent === agent && t.status === 'completed');
   337→    const hasError = state.traces.some(t => t.agent === agent && t.status === 'error');
   338→    const hasStarted = state.traces.some(t => t.agent === agent && t.status === 'started');
   339→    if (hasError) return 'error';
   340→    if (hasCompleted) return 'completed';
   341→    if (hasStarted) return 'running';
   342→    // For Gate 1 agents (advocate + triage), show running during phase 1
   343→    if (agent === 'patient-advocate' && state.phase === 'running-phase1') return 'running';
   344→    if (agent === 'denial-triage' && state.phase === 'running-phase1') return 'running';
   345→    // For post-Gate 1 agents, show running during phase 2
   346→    if (['policy-research', 'evidence-assembly', 'letter-drafting', 'quality-review'].includes(agent) && state.phase === 'running-phase2') return 'running';
   347→    return 'pending';
   348→  };
   349→
   350→  const provenanceTierColor = (tier: string) => {
   351→    switch (tier) {
   352→      case 'primary_source': return 'bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200';
   353→      case 'secondary_summary': return 'bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200';
   354→      default: return 'bg-gray-100 text-gray-800 dark:bg-gray-900 dark:text-gray-200';
   355→    }
   356→  };
   357→
   358→  return (
   359→    <div className="space-y-6">
   360→      {/* ── Input Section ─────────────────────────────────────────────── */}
   361→      <Card>
   362→        <CardHeader>
   363→          <CardTitle className="flex items-center gap-2">
   364→            <Radio className="h-5 w-5 text-teal-600" />
   365→            Day 6: Decision Trace + HITL Gates + UI Stream
   366→          </CardTitle>
   367→          <CardDescription>
   368→            Full pipeline with live trace streaming, both HITL gates, letter editing with version history
   369→          </CardDescription>
   370→        </CardHeader>
   371→        <CardContent className="space-y-4">
   372→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   373→            <div className="space-y-2">
   374→              <Label className="text-xs font-medium">Sample Denial Letter</Label>
   375→              <Select value={selectedSample} onValueChange={handleSampleSelect}>
   376→                <SelectTrigger className="text-sm">
   377→                  <SelectValue placeholder="Select sample..." />
   378→                </SelectTrigger>
   379→                <SelectContent>
   380→                  {SAMPLE_LETTERS.map((s, i) => (
   381→                    <SelectItem key={i} value={String(i)}>{s.label}</SelectItem>
   382→                  ))}
   383→                </SelectContent>
   384→              </Select>
   385→            </div>
   386→            <div className="space-y-2">
   387→              <Label className="text-xs font-medium">Payer</Label>
   388→              <Select value={payer} onValueChange={setPayer}>
   389→                <SelectTrigger className="text-sm">
   390→                  <SelectValue />
   391→                </SelectTrigger>
   392→                <SelectContent>
   393→                  <SelectItem value="Medicare">Medicare</SelectItem>
   394→                  <SelectItem value="UnitedHealthcare">UnitedHealthcare</SelectItem>
   395→                  <SelectItem value="Aetna">Aetna</SelectItem>
   396→                </SelectContent>
   397→              </Select>
   398→            </div>
   399→          </div>
   400→          <Textarea
   401→            value={denialText}
   402→            onChange={(e) => setDenialText(e.target.value)}
   403→            placeholder="Paste denial letter text here..."
   404→            className="min-h-[120px] text-sm font-mono"
   405→          />
   406→          <div className="flex gap-3 flex-wrap">
   407→            <Button
   408→              onClick={runPhase1}
   409→              disabled={isRunning || state.phase !== 'idle'}
   410→              className="bg-teal-600 hover:bg-teal-700 text-white"
   411→            >
   412→              {isRunning ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Radio className="h-4 w-4 mr-2" />}
   413→              Run Full Pipeline
   414→            </Button>
   415→            <Button
   416→              onClick={runGateTest}
   417→              disabled={gateTestLoading}
   418→              variant="outline"
   419→              className="border-purple-300 text-purple-700 hover:bg-purple-50"
   420→            >
   421→              {gateTestLoading ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Shield className="h-4 w-4 mr-2" />}
   422→              Gate Test
   423→            </Button>
   424→          </div>
   425→        </CardContent>
   426→      </Card>
   427→
   428→      {/* ── Agent Step Progress ────────────────────────────────────────── */}
   429→      {state.phase !== 'idle' && (
   430→        <Card>
   431→          <CardHeader className="pb-3">
   432→            <CardTitle className="text-sm flex items-center gap-2">
   433→              <Zap className="h-4 w-4 text-amber-500" />
   434→              Pipeline Progress
   435→              <Badge variant="outline" className="text-xs ml-auto">
   436→                {state.latencyMs}ms
   437→              </Badge>
   438→            </CardTitle>
   439→          </CardHeader>
   440→          <CardContent>
   441→            <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
   442→              {AGENT_STEPS.map((step) => {
   443→                const status = agentStepStatus(step.agent);
   444→                const Icon = step.icon;
   445→                return (
   446→                  <div
   447→                    key={step.agent}
   448→                    className={`flex flex-col items-center gap-1.5 p-3 rounded-lg border ${
   449→                      status === 'completed' ? 'border-emerald-300 bg-emerald-50 dark:border-emerald-700 dark:bg-emerald-950' :
   450→                      status === 'running' ? 'border-teal-300 bg-teal-50 dark:border-teal-700 dark:bg-teal-950' :
   451→                      status === 'error' ? 'border-red-300 bg-red-50 dark:border-red-700 dark:bg-red-950' :
   452→                      'border-gray-200 bg-gray-50 dark:border-gray-700 dark:bg-gray-900'
   453→                    }`}
   454→                  >
   455→                    <Icon className={`h-5 w-5 ${step.color} ${status === 'running' ? 'animate-pulse' : ''}`} />
   456→                    <span className="text-[10px] font-medium text-center leading-tight">{step.label}</span>
   457→                    {status === 'completed' && <CheckCircle2 className="h-3 w-3 text-emerald-500" />}
   458→                    {status === 'running' && <Loader2 className="h-3 w-3 text-teal-500 animate-spin" />}
   459→                    {status === 'error' && <XCircle className="h-3 w-3 text-red-500" />}
   460→                  </div>
   461→                );
   462→              })}
   463→            </div>
   464→          </CardContent>
   465→        </Card>
   466→      )}
   467→
   468→      {/* ── Live Decision Trace Stream (Figure 14.1) ──────────────────── */}
   469→      {state.traces.length > 0 && (
   470→        <Card>
   471→          <CardHeader className="pb-3">
   472→            <CardTitle className="text-sm flex items-center gap-2">
   473→              <Eye className="h-4 w-4 text-cyan-500" />
   474→              Live Decision Trace
   475→              <Badge variant="outline" className="text-xs">{state.traces.length} events</Badge>
   476→            </CardTitle>
   477→          </CardHeader>
   478→          <CardContent className="space-y-4">
   479→            {/* Figure 14.1 Checklist Format */}
   480→            {state.traceChecklist.length > 0 && (
   481→              <div className="bg-muted/30 rounded-lg p-4 space-y-3 border border-dashed border-muted-foreground/20">
   482→                <p className="text-xs font-medium text-muted-foreground mb-2">Figure 14.1 — Decision Trace Checklist</p>
   483→                {(() => {
   484→                  const agents = [...new Set(state.traceChecklist.map(i => i.agent))];
