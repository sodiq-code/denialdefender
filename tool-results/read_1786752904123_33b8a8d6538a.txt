   132→// ─── Step Indicator ───────────────────────────────────────────────────────
   133→
   134→function StepIndicator({
   135→  steps,
   136→  currentStep,
   137→}: {
   138→  steps: { label: string; icon: typeof FileText }[];
   139→  currentStep: number; // 0 = not started, 1-3 = in progress, 4 = done
   140→}) {
   141→  return (
   142→    <div className="flex items-center gap-1 sm:gap-2 overflow-x-auto pb-2">
   143→      {steps.map((step, idx) => {
   144→        const stepNum = idx + 1;
   145→        const isActive = currentStep === stepNum;
   146→        const isDone = currentStep > stepNum;
   147→        const Icon = step.icon;
   148→
   149→        return (
   150→          <div key={step.label} className="flex items-center shrink-0">
   151→            <div
   152→              className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-all duration-300 ${
   153→                isDone
   154→                  ? 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900 dark:text-emerald-200'
   155→                  : isActive
   156→                  ? 'bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200 ring-2 ring-teal-400'
   157→                  : 'bg-muted/50 text-muted-foreground'
   158→              }`}
   159→            >
   160→              {isDone ? (
   161→                <CheckCircle2 className="h-3.5 w-3.5" />
   162→              ) : isActive ? (
   163→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   164→              ) : (
   165→                <Icon className="h-3.5 w-3.5" />
   166→              )}
   167→              <span className="hidden sm:inline">{step.label}</span>
   168→              <span className="sm:hidden">{stepNum}</span>
   169→            </div>
   170→            {idx < steps.length - 1 && (
   171→              <ChevronRight className="h-3.5 w-3.5 text-muted-foreground mx-0.5 sm:mx-1 shrink-0" />
   172→            )}
   173→          </div>
   174→        );
   175→      })}
   176→    </div>
   177→  );
   178→}

   279→      // Instead, we'll use inline sample text directly.
   280→    } catch {
   281→      // Ignore
   282→    }
   283→
   284→    // Use inline samples for immediate loading (no server round-trip needed)

   288→      setDenialText(sample.text);
   289→      setPayer(sample.payer);
   290→    }
   291→  }, []);

   593→            </CardContent>

   605→                <Badge variant="outline" className="text-[10px]">

   607→                </Badge>

   609→            </CardHeader>
   610→            <CardContent className="space-y-4">

   614→                  {renderAppealWithCitations(result.appealDraft.paragraph, result.citations)}

   661→                  </AccordionContent>

   664→            </CardContent>
   665→          </Card>
   666→
   667→          {/* Gate Status Card */}
   668→          <Card className={result.gatePassed ? 'border-emerald-300 dark:border-emerald-700' : 'border-amber-300 dark:border-amber-700'}>

   677→                  <div>
   678→                    <p className="text-sm font-medium">
   679→                      Gate: {result.gatePassed ? 'PASSED' : 'NOT PASSED'}
   680→                    </p>

   682→                      Requires 3+ citations. Got {result.citations.length}.
   683→                    </p>
   684→                  </div>
   685→                </div>
   686→                <div className="flex items-center gap-2">

   704→        </>
   705→      )}
   706→
   707→      {/* ── Gate Test Results ───────────────────────────────────── */}
   708→      {gateResult && (

   710→          <CardHeader className="pb-3">
   711→            <CardTitle className="text-sm flex items-center gap-2">
   712→              <Shield className="h-4 w-4 text-teal-600" />

   714→              <Badge

   720→              >
   721→                {gateResult.allPassed ? (
   722→                  <><CheckCircle2 className="h-3 w-3 mr-1" />All Passed</>

   724→                  <>{gateResult.passedCount}/{gateResult.totalRuns} Passed</>
   725→                )}
   726→              </Badge>
   727→            </CardTitle>
   728→          </CardHeader>
   729→          <CardContent className="space-y-3">
   730→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 text-xs mb-3">
   731→              <div className="bg-muted/50 rounded p-2.5">
   732→                <span className="text-muted-foreground block mb-0.5">Total Runs</span>
   733→                <p className="font-mono font-semibold">{gateResult.totalRuns}</p>
   734→              </div>
   735→              <div className="bg-muted/50 rounded p-2.5">
   736→                <span className="text-muted-foreground block mb-0.5">Passed</span>
   737→                <p className={`font-mono font-semibold ${gateResult.allPassed ? 'text-emerald-600' : 'text-amber-600'}`}>
   738→                  {gateResult.passedCount}
   739→                </p>
   740→              </div>
   741→              <div className="bg-muted/50 rounded p-2.5">
   742→                <span className="text-muted-foreground block mb-0.5">Total Latency</span>
   743→                <p className="font-mono font-semibold">{gateResult.totalLatencyMs}ms</p>
   744→              </div>
   745→            </div>
   746→
   747→            {gateResult.results.map((r) => (
   748→              <div
   749→                key={r.run}
   750→                className={`flex items-center gap-3 rounded-lg border p-3 ${
   751→                  r.gatePassed
   752→                    ? 'border-emerald-200 dark:border-emerald-800'
   753→                    : 'border-red-200 dark:border-red-800'
   754→                }`}
   755→              >
   756→                <div className="flex items-center justify-center h-6 w-6 rounded-full bg-muted text-xs font-bold shrink-0">
   757→                  {r.run}
   758→                </div>
   759→                <div className="flex-1 min-w-0">
   760→                  <p className="text-xs font-medium truncate">{r.sampleLabel}</p>
   761→                  <p className="text-[10px] text-muted-foreground">
   762→                    {r.citationsCount} citations &middot; {r.latencyMs}ms
   763→                  </p>
   764→                </div>
   765→                {r.gatePassed ? (
   766→                  <CheckCircle2 className="h-4 w-4 text-emerald-500 shrink-0" />
   767→                ) : (
   768→                  <XCircle className="h-4 w-4 text-red-500 shrink-0" />
   769→                )}
   770→              </div>
   771→            ))}
   772→          </CardContent>
   773→        </Card>
   774→      )}
   775→    </div>
   776→  );
   777→}

   779→// ─── Helper: Render appeal paragraph with clickable citation refs ────────

   781→function renderAppealWithCitations(paragraph: string, citations: VerticalSliceCitation[]) {
   782→  // Split on [1], [2], [3] patterns and interleave with clickable spans
   783→  const parts: React.ReactNode[] = [];
   784→  const regex = /\[(\d+)\]/g;
   785→  let lastIndex = 0;
   786→  let match;
   787→
   788→  while ((match = regex.exec(paragraph)) !== null) {
   789→    // Push text before the match
   790→    if (match.index > lastIndex) {
   791→      parts.push(paragraph.slice(lastIndex, match.index));
   792→    }
   793→
   794→    const citationNum = parseInt(match[1]);
   795→    const citation = citations.find(c => c.number === citationNum);
   796→
   797→    parts.push(
   798→      <button
   799→        key={`cite-${match.index}`}
   800→        className="inline-flex items-center gap-0.5 px-1 py-0 rounded bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200 text-xs font-medium hover:bg-teal-200 dark:hover:bg-teal-800 transition-colors"
   801→        title={citation ? `${citation.documentName}${citation.section ? ` — §${citation.section}` : ''}` : `Citation ${citationNum}`}
   802→      >
   803→        [{citationNum}]
   804→        <ExternalLink className="h-2.5 w-2.5" />
   805→      </button>
   806→    );
   807→
   808→    lastIndex = regex.lastIndex;
   809→  }
   810→
   811→  // Push remaining text
   812→  if (lastIndex < paragraph.length) {
   813→    parts.push(paragraph.slice(lastIndex));
   814→  }
   815→
   816→  return parts;
   817→}
