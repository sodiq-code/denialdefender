   467→    overall_policy_assessment: `The policy analysis reveals 2 contradictions between ${carrier}'s medical policy and authoritative clinical guidelines. The strongest argument is that the payer's conservative therapy requirement exceeds the standard of care established by ACR 2024. Combined with the policy gap for this specific diagnosis-procedure combination, these findings substantially support the ${strategy} appeal strategy.`,
   468→  };

   471→function mockCitation(inputData: Record<string, unknown>): CitationResult {
   472→  const evidence = (inputData.evidence ?? {}) as Record<string, unknown>;
   473→  const evidenceItems = (evidence.evidence_items ?? [
   474→    { id: "ev-1", title: "Clinical Practice Guideline", source: "AMA CPT Assistant", provenance_tier: "TIER_4_GUIDELINE", relevance_score: 0.92, year: 2024 },
   475→    { id: "ev-2", title: "Systematic Review of Treatment Efficacy", source: "JAMA", provenance_tier: "TIER_1_SYSTEMATIC_REVIEW", relevance_score: 0.88, year: 2023 },
   476→    { id: "ev-3", title: "RCT of Procedure vs. Usual Care", source: "NEJM", provenance_tier: "TIER_2_RCT", relevance_score: 0.85, year: 2023 },
   477→  ]) as Array<Record<string, unknown>>;
   478→
   479→  const tierWeights: Record<string, number> = {
   480→    TIER_1_SYSTEMATIC_REVIEW: 1.0,
   481→    TIER_2_RCT: 0.8,
   482→    TIER_3_OBSERVATIONAL: 0.6,
   483→    TIER_4_GUIDELINE: 0.7,
   484→    TIER_5_EXPERT_OPINION: 0.4,
   485→  };
   486→
   487→  const tierDistribution: Record<string, number> = {
   488→    TIER_1_SYSTEMATIC_REVIEW: 0,
   489→    TIER_2_RCT: 0,
   490→    TIER_3_OBSERVATIONAL: 0,
   491→    TIER_4_GUIDELINE: 0,
   492→    TIER_5_EXPERT_OPINION: 0,
   493→  };
   494→
   495→  const verified: Array<{
   496→    number: number;
   497→    id: string;
   498→    formatted_citation: string;
   499→    provenance_tier: string;
   500→    tier_weight: number;
   501→    relevance_score: number;
   502→    combined_score: number;
   503→    year: number;
   504→    source_type: string;
   505→    doi: string | null;
   506→    pmid: string | null;
   507→    verified: boolean;
   508→    verification_note: string;
   509→  }> = [];
   510→  let idx = 0;
   511→
   512→  for (const item of evidenceItems) {
   513→    idx++;
   514→    const tier = (item.provenance_tier as string) ?? "TIER_5_EXPERT_OPINION";
   515→    const relevance = (item.relevance_score as number) ?? 0.5;
   516→    const year = (item.year as number) ?? 2023;
   517→    const weight = tierWeights[tier] ?? 0.4;
   518→    const recencyBonus = Math.max(-0.15, (year - 2023) * 0.05);
   519→    const combined = Math.min(1.0, Math.round((weight * relevance + recencyBonus) * 1000) / 1000);
   520→
   521→    tierDistribution[tier] = (tierDistribution[tier] ?? 0) + 1;
   522→
   523→    verified.push({
   524→      number: idx,
   525→      id: (item.id as string) ?? `ev-${idx}`,
   526→      formatted_citation: `${item.source ?? "Unknown"}. ${item.title ?? "Untitled"}. ${year}.`,
   527→      provenance_tier: tier,
   528→      tier_weight: weight,
   529→      relevance_score: relevance,
   530→      combined_score: combined,
   531→      year,
   532→      source_type: tier.includes("TIER_4") ? "guideline" : "journal",
   533→      doi: null,
   534→      pmid: null,
   535→      verified: true,
   536→      verification_note: "Citation format verified; provenance tier confirmed",
   537→    });
   538→  }
   539→
   540→  // Also add policy citations
   541→  const policy = (inputData.policy ?? {}) as Record<string, unknown>;
   542→  const policyRefs = (policy.policy_references ?? []) as Array<Record<string, unknown>>;
   543→  for (const polRef of policyRefs) {
   544→    idx++;
   545→    verified.push({
   546→      number: idx,
   547→      id: `pol-${idx}`,
   548→      formatted_citation: `${polRef.title ?? "Policy Reference"}. Section: ${polRef.section ?? "N/A"}.`,
   549→      provenance_tier: "TIER_4_GUIDELINE",
   550→      tier_weight: 0.7,
   551→      relevance_score: 0.80,
   552→      combined_score: 0.56,
   553→      year: 2024,
   554→      source_type: "regulation",
   555→      doi: null,
   556→      pmid: null,
   557→      verified: true,
   558→      verification_note: "Policy reference included; direct link may be available",
   559→    });
   560→    tierDistribution["TIER_4_GUIDELINE"]++;
   561→  }
   562→
   563→  const avgCombined = verified.length > 0
   564→    ? verified.reduce((s, v) => s + v.combined_score, 0) / verified.length
   565→    : 0;
   566→
   567→  let quality = "weak";
   568→  if (avgCombined >= 0.8) quality = "excellent";
   569→  else if (avgCombined >= 0.6) quality = "good";
   570→  else if (avgCombined >= 0.4) quality = "adequate";
   571→
   572→  return {
   573→    verified_citations: verified,
   574→    tier_distribution: tierDistribution,
   575→    overall_citation_quality: quality,
   576→    recommendations: [
   577→      "Consider adding DOI/PMID references for journal citations",
   578→      "Policy references should include direct URLs where available",
   579→      "All citations meet minimum provenance standards for appeal",
   580→    ],
   581→  };
   582→}

   584→// ─── Inline Workflow Engine ─────────────────────────────────────────────
   585→
   586→/**
   587→ * Run the full 8-agent appeal workflow inline (inside Next.js).
   588→ *
   589→ * Produces output identical to the external agent-fleet mini-service's
   590→ * mock mode. Includes small artificial delays per step to simulate
   591→ * agent execution time.
   592→ *
   593→ * Pipeline:
   594→ *   triage → coder → policy → evidence → citation → draft → review → HITL gate
   595→ */
   596→export async function runInlineWorkflow(request: WorkflowRequest): Promise<WorkflowResult> {
   597→  const caseId = request.case_id ?? randomUUID();
   598→  const workflowId = randomUUID();
   599→  const denial = getDenial(request);
   600→  const startTime = nowISO();
   601→  const startMs = Date.now();
   602→
   603→  // ── Step 1: Triage ──────────────────────────────────────────────────
   604→  await stepDelay();
   605→  const triageResult = mockTriage(request);
   606→  const decisionTraces: DecisionTrace[] = [
   607→    {
   608→      step: 1,
   609→      agent: "triage",
   610→      timestamp: nowISO(),
   611→      result_summary: {
   612→        classification: triageResult.classification,
   613→        confidence: triageResult.confidence,
   614→        strategy: triageResult.strategy,
   615→      },
   616→    },
   617→  ];
   618→
   619→  // If NOT_APPEALABLE, short-circuit and flag for human review
   620→  if (triageResult.classification === "NOT_APPEALABLE") {
   621→    const elapsed = Math.round((Date.now() - startMs) / 1000 * 1000) / 1000;
   622→    return {
   623→      case_id: caseId,
   624→      workflow_id: workflowId,
   625→      status: "needs_review",
   626→      triage: triageResult,
   627→      workflow_stopped_at: "triage",
   628→      stop_reason: "Denial classified as NOT_APPEALABLE — requires human judgment",
   629→      decision_traces: decisionTraces,
   630→      hitl_gate: {
   631→        gate_type: "gate_1",
   632→        status: "pending_approval",
   633→        content: "Triage classified this denial as NOT_APPEALABLE. Human review required to determine if appeal should proceed.",
   634→      },
   635→      _trace: {
   636→        agent: "orchestrator",
   637→        trace_id: randomUUID(),
   638→        elapsed_seconds: elapsed,
   639→        timestamp: startTime,
   640→      },
   641→    };
   642→  }
   643→
   644→  // ── Step 2: Coder ───────────────────────────────────────────────────
   645→  await stepDelay();
   646→  const coderInput = { denial, patient_context: request.patient_context ?? {} };
   647→  const coderResult = mockCoder(coderInput);
   648→  decisionTraces.push({
   649→    step: 2,
   650→    agent: "coder",
   651→    timestamp: nowISO(),
   652→    result_summary: {
   653→      validation_result: coderResult.validation_result,
   654→      coding_action_required: coderResult.coding_action_required,
   655→    },
   656→  });
   657→
   658→  // ── Step 3: Policy ──────────────────────────────────────────────────
   659→  await stepDelay();
   660→  const policyInput = { denial, patient_context: request.patient_context ?? {}, triage: triageResult, coding: coderResult };
   661→  const policyResult = mockPolicy(policyInput);
   662→  decisionTraces.push({
   663→    step: 3,
   664→    agent: "policy",
   665→    timestamp: nowISO(),
   666→    result_summary: {
   667→      contradictions_count: policyResult.contradictions_found.length,
   668→      patient_meets_criteria: policyResult.patient_meets_criteria,
   669→    },
   670→  });
   671→
   672→  // ── Step 4: Evidence ────────────────────────────────────────────────
   673→  await stepDelay();
   674→  const evidenceInput = { denial, patient_context: request.patient_context ?? {}, triage: triageResult };
   675→  const evidenceResult = mockEvidence(evidenceInput);
   676→  decisionTraces.push({
   677→    step: 4,
   678→    agent: "evidence",
   679→    timestamp: nowISO(),
   680→    result_summary: {
   681→      evidence_count: evidenceResult.evidence_items.length,
   682→      overall_strength: evidenceResult.overall_evidence_strength,
   683→    },
   684→  });
   685→
   686→  // ── Step 5: Citation ────────────────────────────────────────────────
   687→  await stepDelay();
   688→  const citationInput = { evidence: evidenceResult, policy: policyResult };
   689→  const citationResult = mockCitation(citationInput);
   690→  decisionTraces.push({
   691→    step: 5,
   692→    agent: "citation",
   693→    timestamp: nowISO(),
   694→    result_summary: {
   695→      verified_count: citationResult.verified_citations.length,
   696→      overall_quality: citationResult.overall_citation_quality,
   697→    },
   698→  });
   699→
   700→  // ── Steps 6-8: Draft → Review → (revise if needed) ─────────────────
   701→  let draftResult: DraftResult = {} as DraftResult;
   702→  let reviewResult: ReviewResult = {} as ReviewResult;
   703→
   704→  for (let loop = 0; loop < MAX_REVISION_LOOPS; loop++) {
   705→    // Step 6: Draft
   706→    await stepDelay();
   707→    const draftInput: Record<string, unknown> = {
   708→      case_id: caseId,
   709→      denial,
   710→      patient_context: request.patient_context ?? {},
   711→      triage: triageResult,
   712→      evidence: evidenceResult,
   713→      policy: policyResult,
   714→      citations: citationResult,
   715→      coding: coderResult,
   716→    };
   717→    if (reviewResult.revision_instructions) {
   718→      draftInput.revision_instructions = reviewResult.revision_instructions;
   719→    }
   720→    draftResult = mockDraft(draftInput);
   721→    decisionTraces.push({
   722→      step: 6,
   723→      agent: "drafter",
   724→      timestamp: nowISO(),
   725→      revision_loop: loop,
   726→      result_summary: {
   727→        word_count: draftResult.word_count,
   728→        citations_count: draftResult.citations_used?.length ?? 0,
   729→      },
   730→    });
   731→
   732→    // Step 7: Review
   733→    await stepDelay();
   734→    const reviewInput = { denial, triage: triageResult, evidence: evidenceResult, draft: draftResult };
   735→    reviewResult = mockReviewer(reviewInput);
   736→    decisionTraces.push({
   737→      step: 7,
   738→      agent: "reviewer",
   739→      timestamp: nowISO(),
   740→      revision_loop: loop,
   741→      result_summary: {
   742→        verdict: reviewResult.overall_verdict,
   743→        score: reviewResult.overall_score,
   744→      },
   745→    });
   746→
   747→    // Step 8: Check revision
   748→    if (reviewResult.overall_verdict === "APPROVED" || reviewResult.overall_verdict === "REJECTED") {
   749→      break;
   750→    }
   751→  }
   752→
   753→  const finalStatus = reviewResult.overall_verdict === "NEEDS_REVISION" ? "needs_review" : "completed";
   754→  const elapsed = Math.round((Date.now() - startMs) / 1000 * 1000) / 1000;
   755→
   756→  const hitlGate: HitlGateResult = {
   757→    gate_type: "gate_2",
   758→    status: "pending_approval",
   759→    content: `Appeal letter generated and reviewed. Human approval required before submission. Review verdict: ${reviewResult.overall_verdict ?? "N/A"}, Score: ${reviewResult.overall_score ?? "N/A"}`,
   760→  };
   761→
   762→  return {
   763→    case_id: caseId,
   764→    workflow_id: workflowId,
   765→    status: finalStatus,
   766→    triage: triageResult,
   767→    coder: coderResult,
   768→    policy: policyResult,
   769→    evidence: evidenceResult,
   770→    citation: citationResult,
   771→    draft: draftResult,
   772→    review: reviewResult,
   773→    decision_traces: decisionTraces,
   774→    hitl_gate: hitlGate,
   775→    _trace: {
   776→      agent: "orchestrator",
   777→      trace_id: randomUUID(),
   778→      elapsed_seconds: elapsed,
   779→      timestamp: startTime,
   780→    },
   781→  };
   782→}
