    13→import React, { useState, useCallback } from 'react';
    14→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';

    18→import { Progress } from '@/components/ui/progress';

    21→import {

    28→

    32→  details: string;

    34→

    64→  memoryBankStatus?: string;

    67→

    69→  path: string;
    70→  success: boolean;
    71→  appealLetterLength: number;
    72→  citationCount: number;

   106→// ─── Main Component ────────────────────────────────────────────────────────
   107→
   108→export default function Day7Panel() {
   109→  // Eval state
   110→  const [heldOutCases, setHeldOutCases] = useState<any[]>([]);
   111→  const [evalSnapshot, setEvalSnapshot] = useState<EvalSnapshot | null>(null);
   112→  const [evalRunning, setEvalRunning] = useState(false);
   113→  const [determinismResult, setDeterminismResult] = useState<DeterminismResult | null>(null);
   114→  const [determinismRunning, setDeterminismRunning] = useState(false);
   115→
   116→  // Outcome state
   117→  const [outcomeResult, setOutcomeResult] = useState<OutcomeIngestResult | null>(null);
   118→  const [outcomeRunning, setOutcomeRunning] = useState(false);
   119→
   120→  // Execution paths state
   121→  const [pathResult, setPathResult] = useState<ExecutionPathResult | null>(null);
   122→  const [pathRunning, setPathRunning] = useState(false);

   124→  const [demoRunning, setDemoRunning] = useState(false);
   125→  const [appealLetter, setAppealLetter] = useState('');
   126→
   127→  // ─── Load Held-Out Cases ──────────────────────────────────────────────
   128→
   129→  const loadCases = useCallback(async () => {
   130→    try {
   131→      const res = await fetch('/api/eval');
   132→      const data = await res.json();
   133→      if (data.success) setHeldOutCases(data.cases || []);

   135→      console.error('Failed to load held-out cases:', e);
   136→    }

   140→

   142→    setEvalRunning(true);
   143→    try {
   144→      const res = await fetch('/api/eval', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ saveToDisk: true }) });
   145→      const data = await res.json();
   146→      if (data.success) {
   147→        setEvalSnapshot(data.snapshot);
   148→      }
   149→    } catch (e: any) {
   150→      console.error('Eval failed:', e);
   151→    } finally {
   152→      setEvalRunning(false);
   153→    }
   154→  }, []);
   155→
   156→  // ─── Verify Determinism ───────────────────────────────────────────────
   157→

   159→    setDeterminismRunning(true);
   160→    try {
   161→      const res = await fetch('/api/eval/determinism', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ runs: 2 }) });
   162→      const data = await res.json();
   163→      if (data.success) {
   164→        setDeterminismResult(data);
   165→      }
   166→    } catch (e: any) {
   167→      console.error('Determinism check failed:', e);
   168→    } finally {
   169→      setDeterminismRunning(false);
   170→    }
   171→  }, []);
   172→
   173→  // ─── Ingest Outcomes ──────────────────────────────────────────────────
   174→
   175→  const ingestPublicOutcomes = useCallback(async () => {
   176→    setOutcomeRunning(true);
   177→    try {
   178→      const res = await fetch('/api/outcome-ingest', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ingestPublic: true }) });
   179→      const data = await res.json();
   180→      setOutcomeResult(data);
   181→    } catch (e: any) {
   182→      setOutcomeResult({ success: false });
   183→    } finally {
   184→      setOutcomeRunning(false);
   185→    }
   186→  }, []);
   187→

   189→    setOutcomeRunning(true);
   190→    try {
   191→      const res = await fetch('/api/outcome-ingest', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ingestSynthetic: true, count: 10 }) });
   192→      const data = await res.json();
   193→      setOutcomeResult(data);
   194→    } catch (e: any) {
   195→      setOutcomeResult({ success: false });
   196→    } finally {
   197→      setOutcomeRunning(false);
   198→    }
   199→  }, []);
   200→
   201→  // ─── Test Execution Paths ─────────────────────────────────────────────
   202→
   203→  const testPath = useCallback(async (path: string) => {
   204→    setPathRunning(true);
   205→    try {
   206→      const res = await fetch('/api/execution-paths', {
   207→        method: 'POST',
   208→        headers: { 'Content-Type': 'application/json' },
   209→        body: JSON.stringify({
   210→          path,
   211→          denialText: 'UnitedHealthcare\nClaims Adjudication Department\n\nDATE: January 15, 2026\n\nRE: Denial of Claim — 27447\n\nDENIAL REASON: CO50 — Not medically necessary',
   212→          payer: 'UnitedHealthcare',
   213→          denialCategory: 'medical_necessity',
   214→          cptCode: '27447',
   215→        }),
   216→      });

   218→      if (data.success) {
   219→        setPathResult(data.result);
   220→        setAppealLetter(data.appealLetter || '');
   221→      }
   222→    } catch (e: any) {
   223→      console.error('Path test failed:', e);
   224→    } finally {
   225→      setPathRunning(false);
   226→    }
   227→  }, []);
   228→
   229→  // ─── Test Demo Reliability ────────────────────────────────────────────
   230→

   232→    setDemoRunning(true);
   233→    try {
   234→      const res = await fetch('/api/execution-paths/demo-test', {
   235→        method: 'POST',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify({
   238→          denialText: 'UnitedHealthcare\n\nDENIAL REASON: CO50 — Not medically necessary',
   239→          payer: 'UnitedHealthcare',
   240→          denialCategory: 'medical_necessity',
   241→        }),
   242→      });
   243→      const data = await res.json();
   244→      if (data.success) {
   245→        setDemoReliability(data);
   246→      }
   247→    } catch (e: any) {
   248→      console.error('Demo reliability test failed:', e);
   249→    } finally {

   251→    }
   252→  }, []);
   253→

   255→
   256→  return (
   257→    <div className="space-y-6">
   258→      <div className="flex items-center gap-3">
   259→        <FlaskConical className="h-6 w-6 text-emerald-600" />
   260→        <div>
   261→          <h2 className="text-xl font-semibold">Day 7: Outcome Learning + Eval + Execution Paths</h2>
   262→          <p className="text-sm text-muted-foreground">
   263→            Evaluation harness, outcome ingestion, demo reliability — 3 execution paths
   264→          </p>
   265→        </div>
   266→      </div>
   267→
   268→      <Tabs defaultValue="eval" className="w-full">
   269→        <TabsList className="grid w-full grid-cols-3">

   271→            <BarChart3 className="h-4 w-4 mr-1" />
   272→            Eval Service
   273→          </TabsTrigger>
   274→          <TabsTrigger value="outcome">
   275→            <GitBranch className="h-4 w-4 mr-1" />
   276→            Outcome Learning
   277→          </TabsTrigger>
   278→          <TabsTrigger value="paths">
   279→            <Server className="h-4 w-4 mr-1" />
   280→            Execution Paths
   281→          </TabsTrigger>
   282→        </TabsList>
   283→
   284→        {/* ─── EVAL SERVICE TAB ─────────────────────────────────────────── */}

   286→          {/* Held-Out Cases */}
   287→          <Card>
   288→            <CardHeader className="pb-3">
   289→              <CardTitle className="text-base flex items-center gap-2">
   290→                <Database className="h-4 w-4" />
   291→                10 Held-Out Cases

   316→                          {c.shouldAppeal ? 'Appealable' : 'Not Appealable'}

   326→          {/* Run Eval */}

   331→                Before-Scores Eval Run

   336→            </CardHeader>

   382→                      </div>

   386→                  {/* Per-Case Breakdown */}
   387→                  <div className="space-y-2 mt-4">
