   173→/**
   174→ * Enforce agent permission at runtime.
   175→ * Returns { allowed, reason } — if not allowed, the endpoint MUST return 403.
   176→ */
   177→function enforcePermission(
   178→  agentName: string,
   179→  resource: Resource,
   180→  capability: Capability,
   181→): { allowed: boolean; reason: string } {
   182→  const scope = AGENT_SCOPES[agentName];
   183→  if (!scope) {
   184→    return { allowed: false, reason: `Unknown agent: ${agentName}` };
   185→  }
   186→  const caps = scope.resources[resource];
   187→  if (!caps || !caps.includes(capability)) {
   188→    return {
   189→      allowed: false,
   190→      reason: `DENIED: ${scope.role} does NOT have ${capability} permission on ${resource}. Action blocked at runtime.`,
   191→    };
   192→  }
   193→  return {
   194→    allowed: true,
   195→    reason: `${scope.role} has ${capability} permission on ${resource}`,
   196→  };
   197→}

  1038→/**
  1039→ * Learned context from outcome learning — injected into agent prompts
  1040→ * so agents adjust behavior based on past appeal outcomes.
  1041→ * This closes the learning loop: Outcome → Weight → Prompt → Better Decision.
  1042→ */
  1043→interface LearnedContext {
  1044→  strategySuccessRates?: Record<string, number>; // e.g., { medical_necessity: 0.72, prior_auth: 0.45 }
  1045→  evidenceWeightHints?: Record<string, number>;   // e.g., { "JAMA": 0.92, "NEJM": 0.88 }
  1046→  payerBehaviorNotes?: string[];                  // e.g., ["UnitedHealthcare overrides 42% at redetermination"]
  1047→  categoryOutcomeCount?: number;                  // how many outcomes learned from
  1048→}
  1049→
  1050→function buildLearnedContextSuffix(ctx: LearnedContext | undefined): string {
  1051→  if (!ctx || (Object.keys(ctx).length === 0)) return '';
  1052→  const lines: string[] = ['\n\n--- LEARNED FROM PAST OUTCOMES ---'];
  1053→  if (ctx.strategySuccessRates && Object.keys(ctx.strategySuccessRates).length > 0) {
  1054→    lines.push('Strategy success rates (from past appeal outcomes):');
  1055→    for (const [strategy, rate] of Object.entries(ctx.strategySuccessRates)) {
  1056→      lines.push(`  - ${strategy}: ${(rate * 100).toFixed(0)}% success rate`);
  1057→    }
  1058→    lines.push('Prefer strategies with higher success rates when multiple options exist.');
  1059→  }
  1060→  if (ctx.evidenceWeightHints && Object.keys(ctx.evidenceWeightHints).length > 0) {
  1061→    lines.push('Evidence source weights (learned from outcomes):');
  1062→    for (const [source, weight] of Object.entries(ctx.evidenceWeightHints)) {
  1063→      lines.push(`  - ${source}: weight ${weight.toFixed(2)}`);
  1064→    }
  1065→    lines.push('Prioritize higher-weighted evidence sources in your response.');
  1066→  }
  1067→  if (ctx.payerBehaviorNotes && ctx.payerBehaviorNotes.length > 0) {
  1068→    lines.push('Payer behavior notes:');
  1069→    for (const note of ctx.payerBehaviorNotes) {
  1070→      lines.push(`  - ${note}`);
  1071→    }
  1072→  }
  1073→  if (ctx.categoryOutcomeCount && ctx.categoryOutcomeCount > 0) {
  1074→    lines.push(`(Based on ${ctx.categoryOutcomeCount} past outcome records)`);
  1075→  }
  1076→  return lines.join('\n');
  1077→}

  1079→async function runAgentWithGemini(
  1080→  agentName: string,
  1081→  systemPrompt: string,
  1082→  inputData: Record<string, unknown>,
  1083→  mockFn: (input: Record<string, unknown>) => Record<string, unknown>,
  1084→  learnedContext?: LearnedContext
  1085→): Promise<{ data: Record<string, unknown>; mode: 'live' | 'mock'; tokensUsed?: number; learnedContextUsed: boolean }> {
  1086→  // If learned context exists, inject it into mock responses too
  1087→  const effectivePrompt = systemPrompt + buildLearnedContextSuffix(learnedContext);
  1088→  const ctxUsed = !!learnedContext && Object.keys(learnedContext).length > 0;
  1089→
  1090→  if (MOCK_MODE) {
  1091→    // Even in mock mode, adjust mock output based on learned context
  1092→    const mockData = mockFn(inputData);
  1093→    if (ctxUsed && learnedContext?.strategySuccessRates) {
  1094→      // Adjust mock confidence/strategy based on learned rates
  1095→      const strategy = (mockData as Record<string, unknown>).strategy as string;

  1097→      if (learnedRate !== undefined && (mockData as Record<string, unknown>).estimated_success_rate !== undefined) {
  1098→        // Blend: 70% learned + 30% default (avoid over-reliance on limited data)
  1099→        (mockData as Record<string, unknown>).estimated_success_rate =
  1100→          Math.round((learnedRate * 0.7 + ((mockData as Record<string, unknown>).estimated_success_rate as number) * 0.3) * 100) / 100;
  1101→        (mockData as Record<string, unknown>).learned_from_outcomes = true;
  1102→        (mockData as Record<string, unknown>).outcome_sample_size = learnedContext.categoryOutcomeCount;
  1103→      }
  1104→    }
  1105→    return { data: mockData, mode: 'mock', learnedContextUsed: ctxUsed };
  1106→  }
  1107→

  1117→        if (learnedRate !== undefined && (mockData as Record<string, unknown>).estimated_success_rate !== undefined) {
  1118→          (mockData as Record<string, unknown>).estimated_success_rate =
  1119→            Math.round((learnedRate * 0.7 + ((mockData as Record<string, unknown>).estimated_success_rate as number) * 0.3) * 100) / 100;
  1120→          (mockData as Record<string, unknown>).learned_from_outcomes = true;
  1121→          (mockData as Record<string, unknown>).outcome_sample_size = learnedContext.categoryOutcomeCount;
  1122→        }
  1123→      }
  1124→      return { data: mockData, mode: 'mock', learnedContextUsed: ctxUsed };
  1125→    }
  1126→
  1127→    const userPrompt = JSON.stringify(inputData, null, 2);
  1128→    const response = await llm.generate(userPrompt, {
  1129→      systemPrompt: effectivePrompt,
  1130→      temperature: 0.3,
  1131→      maxTokens: 4096,
  1132→    });
  1133→
  1134→    if (!response.success || !response.content) {
  1135→      console.warn(`[${agentName}] Gemini call failed: ${response.error}, falling back to mock`);
  1136→      // Still apply learned context to mock fallback
  1137→      const mockData = mockFn(inputData);
  1138→      if (ctxUsed && learnedContext?.strategySuccessRates) {
  1139→        const strategy = (mockData as Record<string, unknown>).strategy as string;
  1140→        const learnedRate = learnedContext.strategySuccessRates[strategy?.toLowerCase()];
  1141→        if (learnedRate !== undefined && (mockData as Record<string, unknown>).estimated_success_rate !== undefined) {
  1142→          (mockData as Record<string, unknown>).estimated_success_rate =
  1143→            Math.round((learnedRate * 0.7 + ((mockData as Record<string, unknown>).estimated_success_rate as number) * 0.3) * 100) / 100;
  1144→          (mockData as Record<string, unknown>).learned_from_outcomes = true;
  1145→          (mockData as Record<string, unknown>).outcome_sample_size = learnedContext.categoryOutcomeCount;
  1146→        }
  1147→      }
  1148→      return { data: mockData, mode: 'mock', learnedContextUsed: ctxUsed };
  1149→    }
  1150→
  1151→    // Parse JSON from response (handle ```json blocks)
  1152→    let jsonStr = response.content.trim();
  1153→    if (jsonStr.startsWith('```json')) {
  1154→      jsonStr = jsonStr.slice(7);
  1155→    }
  1156→    if (jsonStr.startsWith('```')) {
  1157→      jsonStr = jsonStr.slice(3);
  1158→    }
  1159→    if (jsonStr.endsWith('```')) {
  1160→      jsonStr = jsonStr.slice(0, -3);

  1165→    // Mark that learned context was used in this Gemini call
  1166→    if (ctxUsed) {
  1167→      data.learned_from_outcomes = true;
  1168→      data.outcome_sample_size = learnedContext?.categoryOutcomeCount;
  1169→    }
  1170→    console.log(`[${agentName}] Gemini call succeeded (${response.tokensUsed} tokens, learnedCtx=${ctxUsed})`);
  1171→    return { data, mode: 'live', tokensUsed: response.tokensUsed, learnedContextUsed: ctxUsed };
  1172→  } catch (err) {
  1173→    console.warn(`[${agentName}] Gemini parsing failed: ${err}, falling back to mock`);
  1174→    // Still apply learned context to mock fallback
  1175→    const mockData = mockFn(inputData);
  1176→    if (ctxUsed && learnedContext?.strategySuccessRates) {
  1177→      const strategy = (mockData as Record<string, unknown>).strategy as string;
  1178→      const learnedRate = learnedContext.strategySuccessRates[strategy?.toLowerCase()];
  1179→      if (learnedRate !== undefined && (mockData as Record<string, unknown>).estimated_success_rate !== undefined) {

  1183→        (mockData as Record<string, unknown>).outcome_sample_size = learnedContext.categoryOutcomeCount;
  1184→      }
  1185→    }
  1186→    return { data: mockData, mode: 'mock', learnedContextUsed: ctxUsed };
  1187→  }

  1340→async function checkGcpStatus(): Promise<Record<string, unknown>> {
  1341→  const hasGcpCreds = !!process.env.GOOGLE_APPLICATION_CREDENTIALS || !!process.env.GCP_SERVICE_ACCOUNT_KEY;
  1342→  const firestoreStatus: Record<string, unknown> = { available: false, message: "Not configured" };
  1343→  const pubsubStatus: Record<string, unknown> = { available: false, message: "Not configured", topics: [] as string[] };
  1344→
  1345→  // Always try GCP path first (even without explicit creds — Cloud Run has metadata server)
  1346→  const accessToken = await getGcpAccessToken();
  1347→  const isOnGcp = !!accessToken;
  1348→
  1349→  if (isOnGcp || hasGcpCreds) {
  1350→    // ── Real GCP path: try actual Firestore & Pub/Sub REST APIs with auth ──────
  1351→    const authHeaders: Record<string, string> = accessToken
  1352→      ? { Authorization: `Bearer ${accessToken}` }
  1353→      : {};
  1354→
  1355→    try {
  1356→      // Firestore: Check if the (default) database exists using the management API
  1357→      const firestoreUrl = `https://firestore.googleapis.com/v1/projects/${GCP_PROJECT_ID}/databases/(default)`;
  1358→      const firestoreResp = await fetch(firestoreUrl, {
  1359→        method: "GET",
  1360→        headers: authHeaders,
  1361→        signal: AbortSignal.timeout(5000),
  1362→      }).catch(() => null);
  1363→      if (firestoreResp && firestoreResp.ok) {
  1364→        firestoreStatus.available = true;
  1365→        firestoreStatus.message = "Firestore (Cloud Firestore) connected";
  1366→      } else if (firestoreResp && firestoreResp.status === 404) {
  1367→        // Database doesn't exist yet — Firestore is enabled but (default) DB not created
  1368→        // In native mode, the DB is created on first write. This is still "available".
  1369→        firestoreStatus.available = true;
  1370→        firestoreStatus.message = "Firestore enabled (database will be created on first write)";
  1371→      } else if (firestoreResp) {
  1372→        firestoreStatus.available = false;
  1373→        firestoreStatus.message = `Firestore REST API returned status ${firestoreResp.status}`;
  1374→      } else {
  1375→        firestoreStatus.message = "Firestore REST API unreachable (network error)";
  1376→      }
  1377→    } catch {
  1378→      firestoreStatus.message = "Firestore check failed (timeout or error)";
  1379→    }
  1380→
  1381→    try {
  1382→      const pubsubUrl = `https://pubsub.googleapis.com/v1/projects/${GCP_PROJECT_ID}/topics`;
  1383→      const pubsubResp = await fetch(pubsubUrl, {
  1384→        method: "GET",
  1385→        headers: authHeaders,
  1386→        signal: AbortSignal.timeout(5000),
  1387→      }).catch(() => null);
  1388→      if (pubsubResp && pubsubResp.ok) {
  1389→        const data = await pubsubResp.json() as { topics?: Array<{ name: string }> };
  1390→        pubsubStatus.available = true;
  1391→        pubsubStatus.message = `Pub/Sub connected (${(data.topics ?? []).length} topics)`;
  1392→        pubsubStatus.topics = (data.topics ?? []).map((t) => t.name.split("/").pop());
