     1→/**
     2→ * DenialDefender — GEAP Memory Bank (Day 6)
     3→ *
     4→ * Per the Ultimate Blueprint (Section 11 — Memory & State Architecture):
     5→ *   - Long-term memory: Vertex AI Memory Bank → KEEP if stable
     6→ *   - Case state: Firestore
     7→ *   - Evidence & citations: Cloud SQL pgvector
     8→ *
     9→ * Blueprint says: "Vertex AI Memory Bank → KEEP if stable, fall back to Firestore"
    10→ *
    11→ * This implements the three-tier GEAP Memory Bank pattern:
    12→ *
    13→ *   1. Session Memory   — Short-term, per-request (in-memory Map with TTL)
    14→ *   2. Case Memory      — Medium-term, per-case state (Firestore → SQLite fallback)
    15→ *   3. Long-Term Memory — Cross-case patterns, learned weights
    16→ *                         (Vertex AI Memory Bank → Firestore fallback → SQLite fallback)
    17→ *
    18→ * Each tier degrades gracefully: if the primary store is unavailable, the
    19→ * next fallback is used transparently. All operations log which tier/store
    20→ * was actually used for auditability.
    21→ */

   229→  /**
   230→   * Set a session-scoped value with optional TTL (in seconds).
   231→   * Session memory is ephemeral — cleared on process restart.
   232→   */
   233→  setSession(key: string, value: any, ttl?: number): void {
   234→    const expiresAt = ttl ? Date.now() + ttl * 1000 : null;
   235→    this.sessionStore.set(key, { value, expiresAt });
   236→  }
   237→
   238→  /**
   239→   * Get a session-scoped value. Returns null if expired or not found.
   240→   */
   241→  getSession<T>(key: string): T | null {
   242→    const entry = this.sessionStore.get(key);
   243→    if (!entry) return null;
   244→
   245→    // Check TTL
   246→    if (entry.expiresAt && Date.now() > entry.expiresAt) {
   247→      this.sessionStore.delete(key);
   248→      return null;
   249→    }
   250→
   251→    return entry.value as T;
   252→  }
   253→
   254→  /**
   255→   * Clear all session memory entries.
   256→   */
   257→  clearSession(): void {
   258→    this.sessionStore.clear();
   259→  }
   260→
   261→  /**
   262→   * Clean up expired session entries.
   263→   */
   264→  purgeExpiredSession(): number {
   265→    let purged = 0;
   266→    const now = Date.now();
   267→    for (const [key, entry] of this.sessionStore.entries()) {
   268→      if (entry.expiresAt && now > entry.expiresAt) {
   269→        this.sessionStore.delete(key);
   270→        purged++;
   271→      }
   272→    }
   273→    return purged;
   274→  }

   468→        await this.savePatternToFirestore(pattern);

   470→          success: true,

   528→  /**
   529→   * Update outcome weights in long-term memory.
   530→   * This is the core of the Outcome Learning loop.
   531→   *
   532→   * Weight adjustment rules:
   533→   *   - WON: +0.05 to all citations used (capped at 1.0)
   534→   *   - LOST: -0.03 to all citations used (floored at 0.1)
   535→   *   - PARTIAL: +0.02 to citations used
   536→   *   - Per denial_category + payer: strategy-specific adjustments
   537→   */
   538→  async updateOutcomeWeights(weightUpdates: WeightUpdate[]): Promise<MemoryBankResult> {
   539→    const start = Date.now();
   540→
   541→    if (weightUpdates.length === 0) {
   542→      return { success: true, store: 'none', updatesApplied: 0, durationMs: 0 };
   543→    }
   544→
   545→    // Try Vertex AI first
   546→    if (this.longTermStore === 'vertex_ai_memory_bank') {
   547→      try {
   548→        const result = await this.updateWeightsInVertexAI(weightUpdates);
   549→        if (result) {
   550→          return {
   551→            success: true,
   552→            store: 'vertex_ai_memory_bank',
   553→            updatesApplied: weightUpdates.length,
   554→            durationMs: Date.now() - start,
   555→          };
   556→        }
   557→      } catch (e: any) {
   558→        console.warn('[GEAP Memory Bank] Vertex AI weight update failed, trying Firestore:', e.message);
   559→      }
   560→    }
   561→
   562→    // Try Firestore
   563→    if (this.longTermStore === 'vertex_ai_memory_bank' || this.longTermStore === 'firestore_fallback') {
   564→      try {
   565→        const result = await this.updateWeightsInFirestore(weightUpdates);
   566→        if (result) {
   567→          return {
   568→            success: true,
   569→            store: 'firestore_fallback',
   570→            updatesApplied: weightUpdates.length,
   571→            durationMs: Date.now() - start,
   572→          };
   573→        }
   574→      } catch (e: any) {
   575→        console.warn('[GEAP Memory Bank] Firestore weight update failed, trying SQLite:', e.message);
   576→      }
   577→    }
   578→
   579→    // SQLite fallback
   580→    try {
   581→      await this.updateWeightsInSQLite(weightUpdates);
   582→      return {
   583→        success: true,
   584→        store: 'sqlite_fallback',
   585→        updatesApplied: weightUpdates.length,
   586→        durationMs: Date.now() - start,
   587→      };
   588→    } catch (e: any) {
   589→      console.error('[GEAP Memory Bank] All weight update tiers failed:', e.message);
   590→      return {
   591→        success: false,
   592→        store: 'none',
   593→        updatesApplied: 0,
   594→        durationMs: Date.now() - start,
   595→      };
   596→    }
   597→  }

   599→  /**
   600→   * Get outcome weights for a specific denial category + payer.
   601→   */
   602→  async getOutcomeWeights(denialCategory: string, payer: string): Promise<OutcomeWeights> {
   603→    const patterns = await this.getLearnedPatterns({
   604→      patternType: 'strategy_weight',
   605→      denialCategory,
   606→      payer,
   607→    });
   608→
   609→    const weights: Record<string, number> = {};
   610→    let sampleSize = 0;
   611→    let lastUpdated = new Date().toISOString();
   612→
   613→    for (const pattern of patterns) {
   614→      // Merge weights from each pattern
   615→      for (const [key, value] of Object.entries(pattern.data)) {
   616→        if (typeof value === 'number') {
   617→          weights[key] = value;
   618→        }
   619→      }
   620→      sampleSize += pattern.sourceOutcomes.length;
   621→      if (pattern.updatedAt > lastUpdated) {
   622→        lastUpdated = pattern.updatedAt;
   623→      }
   624→    }
   625→
   626→    return {
   627→      denialCategory,
   628→      payer,
   629→      weights,
   630→      sampleSize,
   631→      lastUpdated,
   632→    };
   633→  }

   684→  private async updateWeightsInSQLite(updates: WeightUpdate[]): Promise<void> {
   685→    // Group updates by evidence ID and apply to the evidence table
   686→    for (const update of updates) {
   687→      try {
   688→        await db.evidence.update({
   689→          where: { id: update.evidenceId },
   690→          data: { retrieval_weight: update.newWeight },
   691→        });
   692→      } catch (e: any) {
   693→        // Skip individual failures
   694→        console.warn(`[GEAP Memory Bank] SQLite weight update failed for ${update.evidenceId}:`, e.message);
   695→      }
   696→    }
   697→
   698→    // Also record as a learned pattern
   699→    if (updates.length > 0) {
   700→      const firstUpdate = updates[0];
   701→      const patternId = createHash('sha256')
   702→        .update(`weight-${firstUpdate.evidenceId}-${Date.now()}`)
   703→        .digest('hex')
   704→        .slice(0, 24);
   705→
   706→      const category = firstUpdate.reason.match(/for (\w+)\/(\w+)/);
   707→      const denialCategory = category ? category[1] : 'unknown';
   708→      const payer = category ? category[2] : 'unknown';
   709→
   710→      const weightsMap: Record<string, number> = {};
   711→      for (const u of updates) {
   712→        weightsMap[u.evidenceId] = u.newWeight;
   713→      }
   714→
   715→      await db.learnedPattern.upsert({
   716→        where: { id: patternId },
   717→        create: {
   718→          id: patternId,
   719→          pattern_type: 'strategy_weight',
   720→          denial_category: denialCategory,
   721→          payer,
   722→          data: JSON.stringify(weightsMap),
   723→          confidence: 0.5,
   724→          source_outcomes: JSON.stringify(updates.map((u) => u.outcomeId)),
   725→        },
   726→        update: {
   727→          data: JSON.stringify(weightsMap),
   728→          source_outcomes: JSON.stringify(updates.map((u) => u.outcomeId)),
   729→        },
   730→      });
   731→    }
   732→  }
