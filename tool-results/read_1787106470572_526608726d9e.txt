     4→ * The second layer of the governance vertex. Model Armor runs inside the

   306→// ─── Content Hashing ─────────────────────────────────────────────────────
   307→
   308→/**
   309→ * SHA-256 hash of content for audit deduplication.
   310→ */
   311→export async function hashContent(content: string): Promise<string> {
   312→  const encoder = new TextEncoder();
   313→  const data = encoder.encode(content);
   314→  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
   315→  const hashArray = Array.from(new Uint8Array(hashBuffer));
   316→  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
   317→}

   319→// ─── Content Sanitization ────────────────────────────────────────────────
   320→
   321→/**
   322→ * Remove or neutralize adversarial content from text.
   323→ * Replaces detected threat patterns with [REDACTED] markers.
   324→ */
   325→function sanitizeContent(content: string, threats: InjectionMatch[]): string {
   326→  let sanitized = content;
   327→  for (const threat of threats) {
   328→    if (threat.severity === 'medium' || threat.severity === 'low') {
   329→      // Replace the matched text with a redaction marker
   330→      const marker = `[REDACTED:${threat.type.toUpperCase()}]`;
   331→      // Find and replace occurrences around the position
   332→      const matchLen = threat.matchedText.length;
   333→      if (threat.position >= 0 && threat.position + matchLen <= sanitized.length) {
   334→        sanitized = sanitized.slice(0, threat.position) + marker + sanitized.slice(threat.position + matchLen);
   335→      }
   336→    }
   337→  }
   338→  return sanitized;
   339→}

   341→// ─── Model Armor Scan (Regex-based) ─────────────────────────────────────
   342→
   343→/**
   344→ * Scan content for prompt-injection and jailbreak attempts using regex patterns.
   345→ *
   346→ * This is the LOCAL-DEV / FALLBACK scanner. It runs on retrieved content
   347→ * BEFORE agents process it. The three verdicts:
   348→ *   - ALLOW: content is clean, pass through unchanged
   349→ *   - SANITIZE: content has medium-risk patterns, clean before passing
   350→ *   - BLOCK: content has critical/high-risk patterns, reject entirely
   351→ *
   352→ * Returns ArmorScanResult with verdict, threats, and optional sanitized content.
   353→ */
   354→export function scanContent(content: string): ArmorScanResult {
   355→  const matches: InjectionMatch[] = [];
   356→
   357→  for (const pattern of INJECTION_PATTERNS) {
   358→    const regex = new RegExp(pattern.pattern.source, pattern.pattern.flags);
   359→    let match;
   360→
   361→    while ((match = regex.exec(content)) !== null) {
   362→      const matchedText = match[0].slice(0, 50); // Truncate for safety
   363→
   364→      matches.push({
   365→        type: pattern.type,
   366→        label: pattern.label,
   367→        severity: pattern.severity,
   368→        matchedText,
   369→        position: match.index,
   370→        description: pattern.description,
   371→      });
   372→    }
   373→  }
   374→
   375→  // ── Compute risk score ──
   376→  let riskScore = 0;
   377→  const criticalMatches = matches.filter(m => m.severity === 'critical');
   378→  const highMatches = matches.filter(m => m.severity === 'high');
   379→  const mediumMatches = matches.filter(m => m.severity === 'medium');
   380→  const lowMatches = matches.filter(m => m.severity === 'low');
   381→
   382→  riskScore += Math.min(criticalMatches.length * 50, 100);
   383→  riskScore += Math.min(highMatches.length * 30, 90);
   384→  riskScore += Math.min(mediumMatches.length * 15, 60);
   385→  riskScore += Math.min(lowMatches.length * 5, 20);
   386→  riskScore = Math.min(riskScore, 100);
   387→
   388→  // ── Determine verdict ──
   389→  const hasCriticalOrHigh = criticalMatches.length > 0 || highMatches.length > 0;
   390→  const hasMedium = mediumMatches.length > 0;
   391→
   392→  let verdict: 'ALLOW' | 'SANITIZE' | 'BLOCK';
   393→  let reason: string;
   394→  let sanitizedContent: string | null = null;
   395→
   396→  if (hasCriticalOrHigh) {
   397→    verdict = 'BLOCK';
   398→    reason = `Adversarial content detected: ${criticalMatches.length} critical, ${highMatches.length} high-severity threats. Content blocked from agent processing.`;
   399→  } else if (hasMedium) {
   400→    verdict = 'SANITIZE';
   401→    reason = `Suspicious patterns detected: ${mediumMatches.length} medium-severity. Content sanitized before processing.`;
   402→    sanitizedContent = sanitizeContent(content, matches);
   403→  } else {
   404→    verdict = 'ALLOW';
   405→    reason = matches.length > 0
   406→      ? `Minor patterns detected (${lowMatches.length}) but below risk threshold. Content allowed.`
   407→      : 'Content is clean. No adversarial patterns detected.';
   408→  }
   409→
   410→  return {
   411→    safe: verdict === 'ALLOW',
   412→    threats: matches,
   413→    riskScore,
   414→    verdict,
   415→    sanitizedContent,
   416→    reason,
   417→    timestamp: new Date().toISOString(),
   418→    scanner: 'regex',
   419→  };
   420→}

   422→// ─── GEAP Model Armor Scan ───────────────────────────────────────────────
   423→
   424→/**
   425→ * Scan content using Google's Model Armor API (GEAP).
   426→ *
   427→ * Calls the Google Model Armor sanitize endpoint and normalises the
   428→ * response into our ArmorScanResult shape. If the API call fails for
   429→ * any reason (network error, not deployed, invalid response, etc.),
   430→ * this function throws so the caller can fall back to regex-based scanning.
   431→ *
   432→ * Environment variables required:
   433→ *   GCP_PROJECT_ID        — GCP project identifier
   434→ *   MODEL_ARMOR_POLICY_ID — Model Armor policy resource ID
   435→ *   MODEL_ARMOR_LOCATION  — GCP region (default: us-central1)
   436→ *   MODEL_ARMOR_API_KEY   — Optional API key / access token
   437→ */
   438→export async function scanContentWithGEAP(content: string): Promise<ArmorScanResult> {
   439→  const config = getGEAPConfig();
   440→  if (!config) {
   441→    throw new Error('GEAP Model Armor not configured — GCP_PROJECT_ID or MODEL_ARMOR_POLICY_ID missing');
   442→  }
   443→
   444→  const startTime = Date.now();
   445→  console.log(`[ModelArmor:GEAP] Scanning content (${content.length} chars) via ${config.endpoint}`);
   446→
   447→  // Build request body per GEAP Model Armor API spec
   448→  const requestBody: GEAPSanitizeRequest = {
   449→    content,
   450→    contentType: 'text/plain',
   451→    metadata: {
   452→      source: 'denialdefender',
   453→      scanner: 'model-armor-geap',
   454→    },
   455→  };
   456→
   457→  // Prepare fetch options with optional auth header
   458→  const headers: Record<string, string> = {
   459→    'Content-Type': 'application/json',
   460→  };
   461→
   462→  // Use MODEL_ARMOR_API_KEY or Google default credentials if available
   463→  const apiKey = process.env.MODEL_ARMOR_API_KEY;
   464→  if (apiKey) {
   465→    headers['Authorization'] = `Bearer ${apiKey}`;
   466→  }
   467→  // In GCP environments (Cloud Run, GKE), the metadata server provides
   468→  // an access token. We read it from GOOGLE_OAUTH_ACCESS_TOKEN if set
   469→  // by the Cloud Run metadata server, or from GOOGLE_APPLICATION_CREDENTIALS.
   470→  const accessToken = process.env.GOOGLE_OAUTH_ACCESS_TOKEN;
   471→  if (accessToken && !apiKey) {
   472→    headers['Authorization'] = `Bearer ${accessToken}`;
   473→  }
   474→
   475→  try {
   476→    const response = await fetch(config.endpoint, {
   477→      method: 'POST',
   478→      headers,
   479→      body: JSON.stringify(requestBody),
   480→      signal: AbortSignal.timeout(10_000), // 10s timeout
   481→    });
   482→
   483→    if (!response.ok) {
   484→      const errorBody = await response.text().catch(() => 'unable to read body');
   485→      throw new Error(
   486→        `GEAP Model Armor API returned ${response.status} ${response.statusText}: ${errorBody}`
   487→      );
   488→    }
   489→
   490→    const raw: GEAPRawApiResponse = await response.json();
   491→
   492→    // Check for API-level error in response body
   493→    if (raw.error) {
   494→      throw new Error(
   495→        `GEAP Model Armor API error: ${raw.error.message || 'unknown'} (code: ${raw.error.code})`
   496→      );
   497→    }
   498→
   499→    // Normalise the raw API response to our GEAPSanitizeResponse shape
   500→    const normalized = normalizeGEAPResponse(raw);
   501→
   502→    const elapsedMs = Date.now() - startTime;
   503→    console.log(
   504→      `[ModelArmor:GEAP] Scan complete in ${elapsedMs}ms — verdict: ${normalized.verdict}, ` +
   505→      `findings: ${normalized.findings?.length ?? 0}`
   506→    );
   507→
   508→    // Convert GEAP response to ArmorScanResult
   509→    return geapResponseToScanResult(content, normalized, config.policyId);
   510→  } catch (error) {
   511→    // Re-throw so caller can fall back to regex
   512→    const elapsedMs = Date.now() - startTime;
   513→    console.warn(
   514→      `[ModelArmor:GEAP] Scan failed after ${elapsedMs}ms:`,
   515→      error instanceof Error ? error.message : error
   516→    );
   517→    throw error;
   518→  }
   519→}
   520→
   521→/**
   522→ * Normalise the raw GEAP API response into our standard GEAPSanitizeResponse.
   523→ * Handles both nested (sanitizeResult.*) and flat response shapes.
   524→ */
   525→function normalizeGEAPResponse(raw: GEAPRawApiResponse): GEAPSanitizeResponse {
   526→  // Prefer the nested sanitizeResult if present
   527→  const nested = raw.sanitizeResult;
   528→  if (nested) {
   529→    return {
   530→      verdict: (nested.verdict as GEAPSanitizeResponse['verdict']) || 'ALLOW',
   531→      confidence: nested.confidence,
   532→      findings: nested.findings?.map(f => ({
   533→        detector: f.detector || 'unknown',
   534→        category: f.category || 'unknown',
   535→        confidence: f.confidence ?? 0,
   536→        blocked: f.blocked ?? false,
   537→        explanation: f.explanation,
   538→        matchedText: f.matchedText,
   539→      })),
   540→      sanitizedContent: nested.sanitizedContent,
   541→      explanation: nested.explanation,
   542→    };
   543→  }
   544→
   545→  // Fall back to flat structure
   546→  return {
   547→    verdict: (raw.verdict as GEAPSanitizeResponse['verdict']) || 'ALLOW',
   548→    confidence: raw.confidence,
   549→    findings: raw.findings?.map(f => ({
   550→      detector: f.detector || 'unknown',
   551→      category: f.category || 'unknown',
   552→      confidence: f.confidence ?? 0,
   553→      blocked: f.blocked ?? false,
   554→      explanation: f.explanation,
   555→      matchedText: f.matchedText,
   556→    })),
   557→    sanitizedContent: raw.sanitizedContent,
   558→    explanation: raw.explanation,
   559→  };
   560→}

   562→/**
   563→ * Convert a GEAPSanitizeResponse into our ArmorScanResult format.
   564→ * Maps GEAP findings to InjectionMatch entries for a consistent interface.

   566→function geapResponseToScanResult(
   567→  content: string,
   568→  geapResponse: GEAPSanitizeResponse,
   569→  policyId: string,
   570→): ArmorScanResult {
   571→  // Map GEAP findings to InjectionMatch entries
   572→  const threats: InjectionMatch[] = (geapResponse.findings || []).map((finding, index) => ({
   573→    type: finding.category || finding.detector,
   574→    label: finding.detector,
   575→    severity: finding.blocked ? 'critical' : finding.confidence >= 0.7 ? 'high' : finding.confidence >= 0.4 ? 'medium' : 'low',
   576→    matchedText: finding.matchedText || '',
   577→    position: index, // GEAP doesn't provide position; use ordinal
   578→    description: finding.explanation || `${finding.detector}: ${finding.category}`,
   579→  }));
   580→
   581→  // Derive risk score from GEAP confidence + findings
   582→  let riskScore: number;
   583→  if (geapResponse.confidence !== undefined) {
   584→    riskScore = Math.round(geapResponse.confidence * 100);
   585→  } else {
   586→    // Compute from findings if no top-level confidence
   587→    const blockedCount = threats.filter(t => t.severity === 'critical').length;
   588→    const highCount = threats.filter(t => t.severity === 'high').length;
   589→    const mediumCount = threats.filter(t => t.severity === 'medium').length;
   590→    const lowCount = threats.filter(t => t.severity === 'low').length;
   591→    riskScore = Math.min(
   592→      blockedCount * 50 + highCount * 30 + mediumCount * 15 + lowCount * 5,
   593→      100
   594→    );
   595→  }
   596→
   597→  const verdict = geapResponse.verdict;
   598→  const reason = geapResponse.explanation ||
   599→    `GEAP Model Armor verdict: ${verdict} (${threats.length} findings, risk score ${riskScore})`;
   600→
   601→  return {
   602→    safe: verdict === 'ALLOW',
   603→    threats,
   604→    riskScore,
   605→    verdict,
   606→    sanitizedContent: geapResponse.sanitizedContent || null,
   607→    reason,
   608→    timestamp: new Date().toISOString(),
   609→    scanner: 'geap',
   610→    policyId,
   611→  };
   612→}
   613→
   614→// ─── Model Armor Gate ────────────────────────────────────────────────────
   615→
   616→/**
   617→ * Run the Model Armor gate on retrieved content before agent processing.
   618→ *
   619→ * This is the second layer of defense (after PHI Guard):
   620→ *   1. Scan content — prefer GEAP Model Armor API, fall back to regex
   621→ *   2. Persist scan result to audit log (with scanner type)
   622→ *   3. Emit decision trace event (with scanner type)
   623→ *
   624→ * Scanner selection logic (Task 4):
   625→ *   - If GCP_PROJECT_ID + MODEL_ARMOR_POLICY_ID are set → try GEAP first
   626→ *   - If GEAP call fails or env not configured → fall back to regex
   627→ *   - Audit entries always record which scanner was used
   628→ *
   629→ * Returns the scan result and audit entry.
   630→ */
   631→export async function runModelArmor(
   632→  content: string,
   633→  contentSource: 'retrieved_policy' | 'retrieved_evidence' | 'external_content',
   634→  caseId?: string,
   635→  agentName?: string,
   636→): Promise<{
   637→  result: ArmorScanResult;
   638→  auditId?: string;
   639→  traceEventId?: string;
   640→}> {
   641→  // ── Step 1: Scan content — GEAP with regex fallback ──
   642→  let result: ArmorScanResult;
   643→  let usedFallback = false;
   644→
   645→  if (isGEAPAvailable()) {
   646→    try {
   647→      result = await scanContentWithGEAP(content);
   648→      console.log('[ModelArmor] Used GEAP Model Armor API — scanner: geap');
   649→    } catch (geapError) {
   650→      console.warn(
   651→        '[ModelArmor] GEAP scan failed, falling back to regex:',
   652→        geapError instanceof Error ? geapError.message : geapError
   653→      );
   654→      result = scanContent(content);
   655→      usedFallback = true;
   656→    }
   657→  } else {
   658→    result = scanContent(content);
   659→    console.log('[ModelArmor] GEAP not configured — using regex scanner (local dev mode)');
   660→  }
   661→
   662→  const contentHash = await hashContent(content);
   663→
   664→  // ── Step 2: Persist audit entry (with scanner type) ──
   665→  let auditId: string | undefined;
   666→  try {
   667→    const audit = await db.governanceAudit.create({
   668→      data: {
   669→        case_id: caseId || null,
   670→        component: 'model_armor',
   671→        action: result.verdict === 'BLOCK' ? 'block' : result.verdict === 'SANITIZE' ? 'scan' : 'allow',
   672→        agent_name: agentName || null,
   673→        verdict: result.verdict,
   674→        risk_score: result.riskScore,
   675→        details: JSON.stringify({
   676→          contentSource,
   677→          contentHash,
   678→          scanner: result.scanner,
   679→          policyId: result.policyId || undefined,
   680→          usedFallback,
   681→          threatCount: result.threats.length,
   682→          threatTypes: [...new Set(result.threats.map(t => t.type))],
   683→          threats: result.threats.map(t => ({
   684→            type: t.type,
   685→            severity: t.severity,
   686→            position: t.position,
   687→          })),
   688→          reason: result.reason,
   689→        }),
   690→        content_hash: contentHash,
   691→      },
   692→    });
   693→    auditId = audit.id;
   694→  } catch (error) {
   695→    console.warn('[ModelArmor] Failed to persist audit entry:', error);
   696→  }
   697→
   698→  // ── Step 3: Emit trace event (with scanner type) ──
   699→  let traceEventId: string | undefined;
   700→  if (caseId) {
   701→    try {
   702→      const traceEvent = await emitTraceEvent({
   703→        caseId,
   704→        agent: 'model-armor',
   705→        step: result.verdict === 'BLOCK' ? 'armor_block' : result.verdict === 'SANITIZE' ? 'armor_sanitize' : 'armor_allow',
   706→        status: result.verdict === 'BLOCK' ? 'blocked' : 'completed',
   707→        detail: result.reason,
   708→        timestamp: new Date().toISOString(),
   709→        metadata: {
   710→          scanner: result.scanner,
   711→          policyId: result.policyId || undefined,
   712→          usedFallback,
   713→          riskScore: result.riskScore,
   714→          threatCount: result.threats.length,
   715→          threatTypes: [...new Set(result.threats.map(t => t.type))],
   716→          contentSource,
   717→          contentHash,
   718→        },
   719→      });
   720→      traceEventId = traceEvent.id;
   721→    } catch (error) {
   722→      console.warn('[ModelArmor] Failed to emit trace event:', error);
   723→    }
   724→  }
   725→
   726→  return { result, auditId, traceEventId };
   727→}

   734→export async function getModelArmorAudit(caseId?: string): Promise<GovernanceAuditRow[]> {

   736→    ? { component: 'model_armor', case_id: caseId }
   737→    : { component: 'model_armor' };
   738→
   739→  const entries = await db.governanceAudit.findMany({
   740→    where,
   741→    orderBy: { timestamp: 'desc' },
   742→    take: 100,
   743→  });
   744→
   745→  return entries.map(mapAuditRow);
   746→}

   822→export interface GovernanceAuditRow {
   823→  id: string;
   824→  caseId: string | null;
   825→  component: string;
   826→  action: string;
   827→  agentName: string | null;
   828→  verdict: string;
   829→  riskScore: number;
   830→  details: Record<string, unknown>;
   831→  contentHash: string | null;
   832→  timestamp: string;
   833→}
   834→
   835→function mapAuditRow(e: {
   836→  id: string;
   837→  case_id: string | null;
   838→  component: string;
   839→  action: string;
   840→  agent_name: string | null;
   841→  verdict: string;
   842→  risk_score: number;
   843→  details: string | null;
   844→  content_hash: string | null;
   845→  timestamp: Date;
   846→}): GovernanceAuditRow {
   847→  let parsed: Record<string, unknown> = {};
   848→  try {
   849→    parsed = JSON.parse(e.details || '{}');
   850→  } catch { /* ignore */ }
   851→
   852→  return {
   853→    id: e.id,
   854→    caseId: e.case_id,
   855→    component: e.component,
   856→    action: e.action,
   857→    agentName: e.agent_name,
   858→    verdict: e.verdict,
   859→    riskScore: e.risk_score,
   860→    details: parsed,
   861→    contentHash: e.content_hash,
   862→    timestamp: e.timestamp.toISOString(),
   863→  };
   864→}
