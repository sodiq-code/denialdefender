     6→ *   - Central registration of all 8 DenialDefender agents with full metadata

    23→ *   5. Letter Drafting — compose evidence-backed appeal draft

   409→// ─── Auto-Registration ────────────────────────────────────────────────────
   410→
   411→/**
   412→ * Auto-register all 8 agents on module import.
   413→ * This ensures the registry is populated when the app starts.
   414→ */
   415→function autoRegister(): void {
   416→  if (autoRegistered) return;
   417→  autoRegistered = true;
   418→
   419→  for (const agent of AGENT_CATALOG) {
   420→    registry.set(agent.name, { ...agent });
   421→    registryEvents.push({
   422→      type: 'agent_registered',
   423→      agentName: agent.name,
   424→      timestamp: new Date().toISOString(),
   425→      detail: `${agent.displayName} v${agent.version} registered [${agent.category}]`,
   426→    });
   427→  }
   428→
   429→  REGISTRY_VERSION = '1.0.0';
   430→  console.log(`[AgentRegistry] Auto-registered ${AGENT_CATALOG.length} agents (registry v${REGISTRY_VERSION})`);
   431→}
   432→
   433→// Execute auto-registration on import
   434→autoRegister();

   436→// ─── Registration API ─────────────────────────────────────────────────────
   437→
   438→/**
   439→ * Register an agent in the registry.
   440→ * If an agent with the same name already exists, it will be updated.
   441→ * Emits a registry event to the decision trace.
   442→ */
   443→export function registerAgent(agent: AgentRegistration): void {
   444→  const existing = registry.get(agent.name);
   445→  const isUpdate = !!existing;
   446→
   447→  registry.set(agent.name, { ...agent });
   448→
   449→  const event: RegistryEvent = {
   450→    type: 'agent_registered',
   451→    agentName: agent.name,
   452→    timestamp: new Date().toISOString(),
   453→    detail: isUpdate
   454→      ? `${agent.displayName} updated from v${existing.version} to v${agent.version}`
   455→      : `${agent.displayName} v${agent.version} registered [${agent.category}]`,
   456→  };
   457→  registryEvents.push(event);
   458→
   459→  // Emit trace event (fire and forget)
   460→  emitRegistryTraceEvent(event).catch(() => {
   461→    // Non-critical — trace emission is best-effort
   462→  });
   463→
   464→  console.log(`[AgentRegistry] ${event.detail}`);
   465→}

   476→/**
   477→ * List agents with optional filtering.
   478→ * Supports filtering by category and/or capability.
   479→ */
   480→export function listAgents(filter?: {
   481→  category?: string;
   482→  capability?: string;
   483→}): AgentRegistration[] {
   484→  let agents = Array.from(registry.values());
   485→
   486→  if (filter?.category) {
   487→    agents = agents.filter(a => a.category === filter.category);
   488→  }
   489→
   490→  if (filter?.capability) {
   491→    agents = agents.filter(a => a.capabilities.includes(filter.capability!));
   492→  }
   493→
   494→  return agents;
   495→}
   496→
   497→/**
   498→ * Search for agents that have a specific capability.
   499→ * Returns agents sorted by health status (healthy first).
   500→ */
   501→export function searchByCapability(capability: string): AgentRegistration[] {
   502→  const agents = Array.from(registry.values())
   503→    .filter(a => a.capabilities.includes(capability));
   504→
   505→  // Sort by health status: healthy > degraded > offline
   506→  const healthOrder: Record<HealthStatus, number> = {
   507→    healthy: 0,
   508→    degraded: 1,
   509→    offline: 2,
   510→  };
   511→
   512→  agents.sort((a, b) => healthOrder[a.healthStatus] - healthOrder[b.healthStatus]);
   513→
   514→  return agents;
   515→}

   547→): void {
   548→  const agent = registry.get(name);
   549→  if (!agent) {
   550→    console.warn(`[AgentRegistry] Cannot update health for unknown agent: ${name}`);
   551→    return;
   552→  }
   553→
   554→  agent.healthStatus = status;
   555→  agent.lastExecution = execution;
   556→
   557→  const event: RegistryEvent = {
   558→    type: 'health_updated',
   559→    agentName: name,
   560→    timestamp: new Date().toISOString(),
   561→    detail: `${agent.displayName} health: ${status} (last exec: ${execution.success ? 'success' : 'failure'}, ${execution.latencyMs}ms)`,
   562→  };
   563→  registryEvents.push(event);
   564→
   565→  // Emit trace event (fire and forget)
   566→  emitRegistryTraceEvent(event).catch(() => {
   567→    // Non-critical
   568→  });
   569→}

   571→/**
   572→ * Bulk health status update from pipeline results.
   573→ * Updates all agents based on their execution results.
   574→ */
   575→export function bulkHealthUpdate(results: Array<{
   576→  agentName: string;
   577→  success: boolean;
   578→  latencyMs: number;
   579→}>): void {
   580→  const now = new Date().toISOString();
   581→
   582→  for (const result of results) {
   583→    const status: HealthStatus = result.success
   584→      ? (result.latencyMs < 5000 ? 'healthy' : 'degraded')  // High latency = degraded
   585→      : 'offline';
   586→
   587→    updateHealthStatus(result.agentName, status, {
   588→      timestamp: now,
   589→      success: result.success,
   590→      latencyMs: result.latencyMs,
   591→    });
   592→  }
   593→}

   595→// ─── Version Compatibility ────────────────────────────────────────────────
   596→
   597→/**
   598→ * Check if two agents are version-compatible for communication.
   599→ */
   600→export function areVersionCompatible(
   601→  agentA: string,
   602→  agentB: string,
   603→): { compatible: boolean; requiredRange: string | null } {
   604→  const compatMap = VERSION_COMPATIBILITY[agentA];
   605→  if (!compatMap || !compatMap[agentB]) {
   606→    // No explicit compatibility requirement = compatible by default
   607→    return { compatible: true, requiredRange: null };
   608→  }
   609→
   610→  const requiredRange = compatMap[agentB];
   611→  const agentBReg = registry.get(agentB);
   612→  if (!agentBReg) {
   613→    return { compatible: false, requiredRange };
   614→  }
   615→
   616→  // Simple semver check: ">=X.Y.Z" means agentB version must be >= X.Y.Z
   617→  const match = requiredRange.match(/^>=(\d+\.\d+\.\d+)$/);
   618→  if (match) {
   619→    const minVersion = match[1];
   620→    const compatible = compareSemver(agentBReg.version, minVersion) >= 0;
   621→    return { compatible, requiredRange };
   622→  }
   623→
   624→  // If we can't parse the range, assume compatible
   625→  return { compatible: true, requiredRange };
   626→}

   629→ * Get all version compatibility relationships for an agent.
   630→ */
   631→export function getVersionCompatibility(agentName: string): Record<string, string> {
   632→  return VERSION_COMPATIBILITY[agentName] || {};
   633→}

   637→/**
   638→ * Get a summary of the registry state.
   639→ */
   640→export function getRegistrySummary(): RegistrySummary {
   641→  const agents = Array.from(registry.values());
   642→
   643→  const categories: Record<AgentCategory, number> = {
   644→    core: 0,
   645→    governance: 0,
   646→    learning: 0,
   647→  };
   648→
   649→  const healthSummary: Record<HealthStatus, number> = {
   650→    healthy: 0,
   651→    degraded: 0,
   652→    offline: 0,
   653→  };
   654→
   655→  const agentsByVersion: Record<string, string[]> = {};
   656→
   657→  for (const agent of agents) {
   658→    categories[agent.category]++;
   659→    healthSummary[agent.healthStatus]++;
   660→
   661→    if (!agentsByVersion[agent.version]) {
   662→      agentsByVersion[agent.version] = [];
   663→    }
   664→    agentsByVersion[agent.version].push(agent.displayName);
   665→  }
   666→
   667→  return {
   668→    totalAgents: agents.length,
   669→    categories,
   670→    healthSummary,
   671→    agentsByVersion,
   672→    registryVersion: REGISTRY_VERSION,
   673→    lastUpdated: new Date().toISOString(),
   674→  };
   675→}

   715→// ─── Helper: Semver Comparison ────────────────────────────────────────────
   716→
   717→/**
   718→ * Compare two semver strings.
   719→ * Returns: -1 if a < b, 0 if a === b, 1 if a > b
   720→ */
   721→function compareSemver(a: string, b: string): number {
   722→  const partsA = a.split('.').map(Number);
   723→  const partsB = b.split('.').map(Number);
   724→
   725→  for (let i = 0; i < 3; i++) {
   726→    const pA = partsA[i] || 0;
   727→    const pB = partsB[i] || 0;
   728→    if (pA > pB) return 1;
   729→    if (pA < pB) return -1;
   730→  }
   731→
   732→  return 0;
   733→}

   737→/**
   738→ * Emit a registry event as a decision trace event.
   739→ * Uses a synthetic case ID since registry events are system-level.
   740→ */
   741→async function emitRegistryTraceEvent(event: RegistryEvent): Promise<void> {
   742→  try {
   743→    await emitTraceEvent({
   744→      caseId: 'system:agent-registry',
   745→      agent: 'agent-registry',
   746→      step: event.type,
   747→      status: 'completed',
   748→      detail: event.detail,
   749→      timestamp: event.timestamp,
   750→      metadata: {
   751→        registryEvent: event.type,
   752→        agentName: event.agentName,
   753→      },
   754→    });
   755→  } catch (error) {
   756→    // Non-critical — trace emission is best-effort
   757→    console.warn('[AgentRegistry] Failed to emit trace event:', error);
   758→  }
   759→}
