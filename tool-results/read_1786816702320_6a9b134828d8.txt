   121→// ─── Credibility Score Color ──────────────────────────────────────────────
   122→
   123→function credibilityColor(score: number): string {
   124→  if (score >= 80) return 'text-emerald-600';
   125→  if (score >= 60) return 'text-amber-600';
   126→  if (score >= 40) return 'text-orange-600';
   127→  return 'text-red-600';
   128→}
   129→
   130→function credibilityBg(score: number): string {
   131→  if (score >= 80) return 'bg-emerald-100 text-emerald-800';
   132→  if (score >= 60) return 'bg-amber-100 text-amber-800';
   133→  if (score >= 40) return 'bg-orange-100 text-orange-800';
   134→  return 'bg-red-100 text-red-800';
   135→}
   136→
   137→function provenanceTierColor(tier: string): string {
   138→  switch (tier) {
   139→    case 'primary_source': return 'bg-blue-100 text-blue-800';
   140→    case 'secondary_summary': return 'bg-purple-100 text-purple-800';
   141→    case 'tertiary_commentary': return 'bg-gray-100 text-gray-800';
   142→    default: return 'bg-gray-100 text-gray-800';
   143→  }
   144→}
