   174→// ─── Provenance Tier Badge ────────────────────────────────────────────────
   175→
   176→function ProvenanceBadge({ tier }: { tier: string }) {
   177→  switch (tier) {
   178→    case 'primary_source':
   179→      return <Badge className="bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200 border-teal-300 dark:border-teal-700">Primary Source</Badge>;
   180→    case 'secondary_summary':
   181→      return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200 border-amber-300 dark:border-amber-700">Secondary Summary</Badge>;
   182→    case 'tertiary_commentary':
   183→      return <Badge variant="outline" className="text-muted-foreground">Tertiary Commentary</Badge>;
   184→    default:
   185→      return <Badge variant="outline">{tier}</Badge>;
   186→  }
   187→}
   188→
   189→// ─── Urgency Badge ────────────────────────────────────────────────────────
   190→
   191→function UrgencyBadge({ level }: { level: string }) {
   192→  switch (level) {
   193→    case 'critical':
   194→      return <Badge className="bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200 border-red-300 dark:border-red-700">Critical</Badge>;
   195→    case 'high':
   196→      return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200 border-amber-300 dark:border-amber-700">High</Badge>;
   197→    default:
   198→      return <Badge variant="outline">Standard</Badge>;
   199→  }
   200→}

   202→// ─── Component ────────────────────────────────────────────────────────────
   203→
   204→export function ThreeAgentPipelinePanel() {
   205→  const [denialText, setDenialText] = useState('');
   206→  const [payer, setPayer] = useState('Medicare');
   207→  const [loading, setLoading] = useState(false);
   208→  const [resuming, setResuming] = useState(false);
   209→  const [result, setResult] = useState<PipelineResult | null>(null);
   210→  const [error, setError] = useState<string | null>(null);
   211→
   212→  // Run pipeline
   213→  const handleRun = useCallback(async () => {
   214→    if (!denialText.trim()) return;
   215→    setLoading(true);
   216→    setError(null);
   217→    setResult(null);
   218→
   219→    try {
   220→      const res = await fetch('/api/three-agent-pipeline', {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({ denialText, payer }),
   224→      });
   225→
   226→      if (!res.ok) {
   227→        const data = await res.json();
   228→        throw new Error(data.error || 'Pipeline failed');
   229→      }
   230→
   231→      const data: PipelineResult = await res.json();
   232→      setResult(data);
   233→    } catch (err: unknown) {
   234→      setError(err instanceof Error ? err.message : 'Unknown error');
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  }, [denialText, payer]);
   239→
   240→  // Load sample
   241→  const handleLoadSample = useCallback(async (sampleId: string) => {
   242→    try {
   243→      const res = await fetch('/api/three-agent-pipeline');
   244→      if (res.ok) {
   245→        const data = await res.json();
   246→        const sample = data.samples?.find((s: { id: string }) => s.id === sampleId);
   247→        if (sample) {
   248→          setPayer(sample.payer);
   249→        }
   250→      }
   251→    } catch {
   252→      // Fallback — set payer from local data
   253→      const local = SAMPLES.find(s => s.id === sampleId);
   254→      if (local) setPayer(local.payer);
   255→    }
   256→
   257→    // Load sample denial text
   258→    const sampleTexts: Record<string, string> = {
   259→      'sample-1': `Medicare\nClaims Adjudication Department\n\nDATE: March 4, 2026\n\nRE: Denial of Claim — 27447 (Total Knee Arthroplasty)\n\nDear Provider,\n\nThis letter is to inform you that the claim submitted for the above-referenced service has been denied.\n\nDENIAL REASON: CO50 — Non-covered services because it is not deemed medically necessary\n\nPAYER STATEMENT: The requested service is not medically necessary for the diagnosed condition. Conservative treatment options have not been adequately documented as exhausted.\n\nPROCEDURE: 27447 — Total Knee Arthroplasty\nDIAGNOSIS: M17.11 — Primary osteoarthritis, right knee\nAMOUNT DENIED: $34,250.00\n\nAPPEAL RIGHTS: You have the right to appeal this denial within 120 days of the date of this notice.\n\nAPPEAL DEADLINE: July 2, 2026`,
   260→      'sample-2': `UnitedHealthcare\nPrior Authorization Department\n\nDATE: February 28, 2026\n\nRE: Denial of Claim — 70553 (MRI Brain With and Without Contrast)\n\nDear Provider,\n\nThis letter is to inform you that the claim submitted for the above-referenced service has been denied.\n\nDENIAL REASON: CO197 — Precertification/authorization/notification/pre-treatment absent\n\nPAYER STATEMENT: Precertification was not obtained prior to service delivery.\n\nPROCEDURE: 70553 — MRI brain with and without contrast\nDIAGNOSIS: G43.909 — Migraine, unspecified, not intractable\nAMOUNT DENIED: $2,890.00\n\nAPPEAL RIGHTS: You have the right to appeal this denial within 180 days of the date of this notice.\n\nAPPEAL DEADLINE: August 27, 2026`,
   261→      'sample-3': `Aetna\nClaims Adjudication Department\n\nDATE: March 10, 2026\n\nRE: Denial of Claim — 99213 (Office Visit, Established Patient, Level 3)\n\nDear Provider,\n\nThis letter is to inform you that the claim submitted for the above-referenced service has been denied.\n\nDENIAL REASON: CO4 — The procedure code is inconsistent with the modifier used or is inconsistent with the diagnosis\n\nPAYER STATEMENT: Procedure code 99213 is inconsistent with the submitted diagnosis code.\n\nPROCEDURE: 99213 — Office visit, established patient, level 3\nDIAGNOSIS: K21.0 — Gastro-esophageal reflux disease with esophagitis\nAMOUNT DENIED: $156.00\n\nAPPEAL RIGHTS: You have the right to appeal this denial within 90 days of the date of this notice.\n\nAPPEAL DEADLINE: June 8, 2026`,
   262→    };
   263→    setDenialText(sampleTexts[sampleId] || '');
   264→  }, []);
   265→
   266→  // Gate 1 action
   267→  const handleGate1 = useCallback(async (action: 'approved' | 'rejected') => {
   268→    if (!result?.caseId) return;
   269→    setResuming(true);
   270→    setError(null);
   271→
   272→    try {
   273→      const res = await fetch('/api/three-agent-pipeline/resume', {
   274→        method: 'POST',
   275→        headers: { 'Content-Type': 'application/json' },
   276→        body: JSON.stringify({
   277→          caseId: result.caseId,
   278→          gateStatus: action,
   279→          triageResult: result.triage,
   280→        }),
   281→      });
   282→
   283→      if (!res.ok) {
   284→        const data = await res.json();
   285→        throw new Error(data.error || 'Resume failed');
   286→      }
   287→
   288→      const data: PipelineResult = await res.json();
   289→      setResult(data);
   290→    } catch (err: unknown) {
   291→      setError(err instanceof Error ? err.message : 'Unknown error');
   292→    } finally {
   293→      setResuming(false);
   294→    }
   295→  }, [result]);
   296→
   297→  // Pipeline phase indicator
   298→  const pipelinePhase = result
   299→    ? result.pipelineStatus === 'completed'
   300→      ? 'completed'
   301→      : result.pipelineStatus === 'gate1_rejected'
   302→        ? 'rejected'
   303→        : result.gate1.status === 'pending'
   304→          ? 'awaiting_gate1'
   305→          : 'running'
   306→    : 'idle';
   307→
   308→  return (
   309→    <div className="space-y-6">
   310→      {/* ── Input Section ─────────────────────────────────────── */}
   311→      <Card>
   312→        <CardHeader className="pb-3">
   313→          <CardTitle className="flex items-center gap-2 text-base">
   314→            <Shield className="h-5 w-5 text-emerald-600" />
   315→            Three-Agent Pipeline
   316→          </CardTitle>
   317→          <CardDescription>
   318→            Advocate → Triage → [Gate 1] → Policy Research
   319→          </CardDescription>
   320→        </CardHeader>
   321→        <CardContent className="space-y-4">
   322→          {/* Sample selector */}
   323→          <div className="flex flex-wrap gap-2">
   324→            <span className="text-sm text-muted-foreground self-center">Samples:</span>
   325→            {SAMPLES.map(sample => (
   326→              <Button
   327→                key={sample.id}
   328→                variant="outline"
   329→                size="sm"
   330→                onClick={() => handleLoadSample(sample.id)}
   331→                className="text-xs"
   332→              >
   333→                {sample.label}
   334→              </Button>
   335→            ))}
   336→          </div>
   337→
   338→          {/* Payer selector */}
   339→          <div className="flex flex-col sm:flex-row gap-3">
   340→            <div className="w-full sm:w-48">
   341→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Payer</label>
   342→              <Select value={payer} onValueChange={setPayer}>
   343→                <SelectTrigger>
   344→                  <SelectValue />
   345→                </SelectTrigger>
   346→                <SelectContent>
   347→                  {PAYERS.map(p => (
   348→                    <SelectItem key={p} value={p}>{p}</SelectItem>
   349→                  ))}
   350→                </SelectContent>
   351→              </Select>
   352→            </div>
   353→          </div>
   354→
   355→          {/* Denial text */}
   356→          <div>
   357→            <label className="text-xs font-medium text-muted-foreground mb-1 block">Denial Letter</label>
   358→            <Textarea
   359→              value={denialText}
   360→              onChange={e => setDenialText(e.target.value)}
   361→              placeholder="Paste the denial letter text here..."
   362→              className="min-h-40 font-mono text-sm"
   363→            />
   364→          </div>
   365→
   366→          {/* Run button */}
   367→          <Button
   368→            onClick={handleRun}
   369→            disabled={loading || !denialText.trim()}
   370→            className="w-full sm:w-auto"
   371→          >
   372→            {loading ? (
   373→              <>
   374→                <Loader2 className="h-4 w-4 animate-spin mr-2" />
   375→                Running Pipeline...
   376→              </>
   377→            ) : (
   378→              <>
   379→                <Shield className="h-4 w-4 mr-2" />
   380→                Run Three-Agent Pipeline
   381→              </>
   382→            )}
   383→          </Button>
   384→
   385→          {error && (
   386→            <div className="flex items-center gap-2 text-sm text-red-600 dark:text-red-400">
   387→              <AlertCircle className="h-4 w-4" />
   388→              {error}
   389→            </div>
   390→          )}
   391→        </CardContent>
   392→      </Card>
   393→
   394→      {/* ── Pipeline Status Indicator ─────────────────────────── */}
   395→      {result && (
   396→        <Card>
   397→          <CardContent className="py-3">
   398→            <div className="flex items-center justify-between flex-wrap gap-2">
   399→              <div className="flex items-center gap-2">
   400→                <span className="text-sm font-medium">Pipeline:</span>
   401→                {pipelinePhase === 'idle' && <Badge variant="outline">Idle</Badge>}
   402→                {pipelinePhase === 'running' && (
   403→                  <Badge className="bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200">
   404→                    <Loader2 className="h-3 w-3 animate-spin mr-1" /> Running
   405→                  </Badge>
   406→                )}
   407→                {pipelinePhase === 'awaiting_gate1' && (
   408→                  <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200">
   409→                    <Clock className="h-3 w-3 mr-1" /> Awaiting Gate 1
   410→                  </Badge>
   411→                )}
   412→                {pipelinePhase === 'rejected' && (
   413→                  <Badge className="bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200">
   414→                    <XCircle className="h-3 w-3 mr-1" /> Gate 1 Rejected
   415→                  </Badge>
   416→                )}
   417→                {pipelinePhase === 'completed' && (
   418→                  <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900 dark:text-emerald-200">
   419→                    <CheckCircle2 className="h-3 w-3 mr-1" /> Completed
   420→                  </Badge>
   421→                )}
   422→              </div>
   423→              <div className="flex items-center gap-3 text-xs text-muted-foreground">
   424→                {result.caseId && <span>Case: {result.caseId.slice(0, 8)}...</span>}
   425→                <span>{result.latencyMs}ms</span>
   426→              </div>
   427→            </div>
   428→
   429→            {/* Phase progress bar */}
   430→            <div className="flex items-center gap-1 mt-3">
   431→              {[
   432→                { label: 'Advocate', active: true, color: 'rose' },
   433→                { label: 'Triage', active: true, color: 'teal' },
   434→                { label: 'Gate 1', active: result.gate1.status !== 'pending' || pipelinePhase === 'awaiting_gate1', color: result.gate1.status === 'approved' ? 'emerald' : result.gate1.status === 'rejected' ? 'red' : 'amber' },
   435→                { label: 'Policy', active: pipelinePhase === 'completed', color: 'emerald' },
   436→              ].map((phase, idx, arr) => (
   437→                <div key={phase.label} className="flex items-center">
   438→                  <div className={`px-2 py-0.5 rounded text-[10px] font-medium ${
   439→                    phase.active
   440→                      ? phase.color === 'rose' ? 'bg-rose-100 text-rose-700 dark:bg-rose-900 dark:text-rose-300'
   441→                        : phase.color === 'teal' ? 'bg-teal-100 text-teal-700 dark:bg-teal-900 dark:text-teal-300'
   442→                          : phase.color === 'emerald' ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900 dark:text-emerald-300'
   443→                            : phase.color === 'red' ? 'bg-red-100 text-red-700 dark:bg-red-900 dark:text-red-300'
   444→                              : 'bg-amber-100 text-amber-700 dark:bg-amber-900 dark:text-amber-300'
   445→                      : 'bg-gray-100 text-gray-400 dark:bg-gray-800 dark:text-gray-600'
   446→                  }`}>
   447→                    {phase.label}
   448→                  </div>
   449→                  {idx < arr.length - 1 && <ChevronDown className="h-3 w-3 text-muted-foreground mx-0.5 rotate-[-90deg]" />}
   450→                </div>
   451→              ))}
   452→            </div>
   453→          </CardContent>
   454→        </Card>
   455→      )}
   456→
   457→      {/* ── Step 1: Patient Advocate ──────────────────────────── */}
   458→      {result && (
   459→        <Card className="border-rose-200 dark:border-rose-800">
   460→          <CardHeader className="pb-3">
   461→            <CardTitle className="flex items-center gap-2 text-base">
   462→              <Heart className="h-5 w-5 text-rose-500" />
   463→              Patient Advocate
   464→              <Badge variant="outline" className="ml-auto text-xs">
   465→                {result.advocate.caseFraming.urgencyLevel === 'critical' ? '🚨' : result.advocate.caseFraming.urgencyLevel === 'high' ? '⚡' : '📋'}
   466→              </Badge>
   467→            </CardTitle>
   468→          </CardHeader>
   469→          <CardContent className="space-y-3">
   470→            {/* Patient Summary */}
   471→            <div>
   472→              <label className="text-xs font-medium text-muted-foreground">Patient Summary</label>
   473→              <p className="text-sm mt-0.5">{result.advocate.caseFraming.patientSummary}</p>
   474→            </div>
   475→
   476→            {/* Urgency + Deadline */}
   477→            <div className="flex flex-wrap gap-3">
   478→              <div>
   479→                <label className="text-xs font-medium text-muted-foreground">Urgency</label>
   480→                <div className="mt-0.5"><UrgencyBadge level={result.advocate.caseFraming.urgencyLevel} /></div>
   481→              </div>
   482→              <div>
   483→                <label className="text-xs font-medium text-muted-foreground">Deadline</label>
   484→                <p className="text-sm mt-0.5">
   485→                  {result.advocate.caseFraming.deadline
   486→                    ? `${result.advocate.caseFraming.deadline}${result.advocate.caseFraming.deadlineDaysRemaining !== null ? ` (${result.advocate.caseFraming.deadlineDaysRemaining}d remaining)` : ''}`
   487→                    : 'Not found'}
   488→                </p>
   489→              </div>
   490→            </div>
   491→
   492→            {/* Denial Impact */}
   493→            <div>
   494→              <label className="text-xs font-medium text-muted-foreground">Denial Impact</label>
   495→              <p className="text-sm mt-0.5">{result.advocate.caseFraming.denialImpact}</p>
   496→            </div>
   497→
   498→            {/* Recommended Actions */}
   499→            <div>
   500→              <label className="text-xs font-medium text-muted-foreground">Recommended Actions</label>
   501→              <ul className="mt-1 space-y-1">
   502→                {result.advocate.caseFraming.recommendedActions.map((action, i) => (
   503→                  <li key={i} className="flex items-start gap-1.5 text-xs text-muted-foreground">
   504→                    <span className="text-rose-500 mt-0.5">•</span>
   505→                    <span>{action}</span>
   506→                  </li>
   507→                ))}
   508→              </ul>
   509→            </div>
   510→
   511→            {/* Empathetic Note */}
   512→            <div className="bg-rose-50 dark:bg-rose-950/30 rounded-md p-3 border border-rose-200 dark:border-rose-800">
   513→              <label className="text-xs font-medium text-rose-600 dark:text-rose-400">Empathetic Note</label>
   514→              <p className="text-sm mt-0.5 text-rose-800 dark:text-rose-200">{result.advocate.empatheticNote}</p>
   515→            </div>
   516→          </CardContent>
   517→        </Card>
   518→      )}
   519→
