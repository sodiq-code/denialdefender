   245→// ─── Provenance Tier Badge ────────────────────────────────────────────────

   247→function ProvenanceBadge({ tier }: { tier: string }) {
   248→  switch (tier) {
   249→    case 'primary_source':
   250→      return <Badge className="bg-teal-100 text-teal-800 dark:bg-teal-900 dark:text-teal-200 border-teal-300 dark:border-teal-700 text-[10px]">Primary</Badge>;
   251→    case 'secondary_summary':
   252→      return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900 dark:text-amber-200 border-amber-300 dark:border-amber-700 text-[10px]">Secondary</Badge>;
   253→    case 'tertiary_commentary':
   254→      return <Badge variant="outline" className="text-[10px] text-muted-foreground">Tertiary</Badge>;
   255→    default:
   256→      return <Badge variant="outline" className="text-[10px]">{tier}</Badge>;
   257→  }
   258→}

   260→// ─── Agent Step ────────────────────────────────────────────────────────────
   261→
   262→interface AgentStepProps {
   263→  name: string;
   264→  icon: React.ReactNode;
   265→  color: string;
   266→  status: 'idle' | 'running' | 'completed' | 'failed' | 'blocked' | 'pending';
   267→  detail?: string;
   268→  latencyMs?: number;
   269→}
   270→
   271→function AgentStep({ name, icon, color, status, detail, latencyMs }: AgentStepProps) {
   272→  const statusIcon = {
   273→    idle: <Clock className="h-3.5 w-3.5 text-muted-foreground" />,
   274→    running: <Loader2 className="h-3.5 w-3.5 animate-spin text-amber-500" />,
   275→    completed: <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500" />,
   276→    failed: <XCircle className="h-3.5 w-3.5 text-red-500" />,
   277→    blocked: <AlertCircle className="h-3.5 w-3.5 text-red-500" />,
   278→    pending: <Clock className="h-3.5 w-3.5 text-amber-500" />,
   279→  }[status];
   280→
   281→  return (
   282→    <div className="flex items-center gap-2 py-1.5">
   283→      <div className={`flex items-center justify-center h-7 w-7 rounded-md ${color}`}>
   284→        {icon}
   285→      </div>
   286→      <div className="flex-1 min-w-0">
   287→        <div className="flex items-center gap-2">
   288→          <span className="text-sm font-medium truncate">{name}</span>
   289→          {statusIcon}
   290→          {latencyMs !== undefined && status === 'completed' && (
   291→            <span className="text-[10px] text-muted-foreground">{latencyMs}ms</span>
   292→          )}
   293→        </div>
   294→        {detail && (
   295→          <p className="text-xs text-muted-foreground truncate">{detail}</p>
   296→        )}
   297→      </div>
   298→    </div>
   299→  );
   300→}

   302→// ─── Component ────────────────────────────────────────────────────────────
   303→
   304→export function SixAgentPipelinePanel() {
   305→  const [denialText, setDenialText] = useState('');
   306→  const [payer, setPayer] = useState('Medicare');
   307→  const [loading, setLoading] = useState(false);
   308→  const [resuming, setResuming] = useState(false);
   309→  const [gateTestLoading, setGateTestLoading] = useState(false);
   310→  const [result, setResult] = useState<PipelineResult | null>(null);
   311→  const [gateTestResult, setGateTestResult] = useState<GateTestResult | null>(null);
   312→  const [error, setError] = useState<string | null>(null);
   313→  const [activeCitation, setActiveCitation] = useState<number | null>(null);
   314→
   315→  // Run pipeline
   316→  const handleRun = useCallback(async () => {
   317→    if (!denialText.trim()) return;
   318→    setLoading(true);
   319→    setError(null);
   320→    setResult(null);
   321→    setGateTestResult(null);
   322→
   323→    try {
   324→      const res = await fetch('/api/six-agent-pipeline', {
   325→        method: 'POST',
   326→        headers: { 'Content-Type': 'application/json' },
   327→        body: JSON.stringify({ denialText, payer }),
   328→      });
   329→
   330→      if (!res.ok) {
   331→        const data = await res.json();
   332→        throw new Error(data.error || 'Pipeline failed');
   333→      }
   334→
   335→      const data: PipelineResult = await res.json();
   336→      setResult(data);
   337→    } catch (err: unknown) {
   338→      setError(err instanceof Error ? err.message : 'Unknown error');
   339→    } finally {
   340→      setLoading(false);
   341→    }
   342→  }, [denialText, payer]);
   343→
   344→  // Load sample
   345→  const handleLoadSample = useCallback((sampleId: string) => {
   346→    const local = SAMPLES.find(s => s.id === sampleId);
   347→    if (local) setPayer(local.payer);
   348→    setDenialText(SAMPLE_TEXTS[sampleId] || '');
   349→  }, []);
   350→
   351→  // Gate 1 action
   352→  const handleGate1 = useCallback(async (action: 'approved' | 'rejected') => {
   353→    if (!result?.caseId) return;
   354→    setResuming(true);
   355→    setError(null);
   356→
   357→    try {
   358→      const res = await fetch('/api/six-agent-pipeline/resume', {
   359→        method: 'POST',
   360→        headers: { 'Content-Type': 'application/json' },
   361→        body: JSON.stringify({
   362→          caseId: result.caseId,
   363→          gateStatus: action,
   364→          triageResult: result.triage,
   365→          advocateResult: result.advocate,
   366→        }),
   367→      });
   368→
   369→      if (!res.ok) {
   370→        const data = await res.json();
   371→        throw new Error(data.error || 'Resume failed');
   372→      }
   373→
   374→      const data: PipelineResult = await res.json();
   375→      setResult(data);
   376→    } catch (err: unknown) {
   377→      setError(err instanceof Error ? err.message : 'Unknown error');
   378→    } finally {
   379→      setResuming(false);
   380→    }
   381→  }, [result]);
   382→
   383→  // Gate test
   384→  const handleGateTest = useCallback(async () => {
   385→    if (!denialText.trim()) return;
   386→    setGateTestLoading(true);
   387→    setError(null);
   388→    setGateTestResult(null);
   389→
   390→    try {
   391→      const res = await fetch('/api/six-agent-pipeline/gate-test', {
   392→        method: 'POST',
   393→        headers: { 'Content-Type': 'application/json' },
   394→        body: JSON.stringify({ denialText, payer }),
   395→      });
   396→
   397→      if (!res.ok) {
   398→        const data = await res.json();
   399→        throw new Error(data.error || 'Gate test failed');
   400→      }
   401→
   402→      const data: GateTestResult = await res.json();
   403→      setGateTestResult(data);
   404→    } catch (err: unknown) {
   405→      setError(err instanceof Error ? err.message : 'Unknown error');
   406→    } finally {
   407→      setGateTestLoading(false);
   408→    }
   409→  }, [denialText, payer]);
   410→
   411→  // Determine agent statuses
   412→  const hasResult = result !== null;
   413→  const gate1Status = result?.gate1.status;
   414→  const hasPolicy = result !== null && result.policyResearch !== null;
   415→  const hasEvidence = result !== null && result.evidenceAssembly !== null;
   416→  const hasDraft = result !== null && result.letterDrafting !== null;
   417→  const hasQuality = result !== null && result.qualityReview !== null;
   418→  const qualityPassed = result?.qualityReview?.overallVerdict === 'PASS';
   419→
   420→  return (
   421→    <div className="space-y-6">
   422→      {/* ── Input Section ─────────────────────────────────────── */}
   423→      <Card>
   424→        <CardHeader className="pb-3">
   425→          <CardTitle className="flex items-center gap-2 text-base">
   426→            <ShieldCheck className="h-5 w-5 text-emerald-600" />
   427→            Six-Agent Pipeline — Day 5
   428→          </CardTitle>
   429→          <CardDescription>
   430→            Advocate → Triage → [Gate 1] → Policy → Evidence → Draft → Quality Review
   431→          </CardDescription>
   432→        </CardHeader>
   433→        <CardContent className="space-y-4">
   434→          {/* Sample selector */}
   435→          <div className="flex flex-wrap gap-2">
   436→            <span className="text-sm text-muted-foreground self-center">Samples:</span>
   437→            {SAMPLES.map(sample => (
   438→              <Button
   439→                key={sample.id}
   440→                variant="outline"
   441→                size="sm"
   442→                onClick={() => handleLoadSample(sample.id)}
   443→                className="text-xs"
   444→              >
   445→                {sample.label}
   446→              </Button>
   447→            ))}
   448→          </div>
   449→
   450→          {/* Payer selector */}
   451→          <div className="flex flex-col sm:flex-row gap-3">
   452→            <div className="w-full sm:w-48">
   453→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Payer</label>
   454→              <Select value={payer} onValueChange={setPayer}>
   455→                <SelectTrigger>
   456→                  <SelectValue />
   457→                </SelectTrigger>
   458→                <SelectContent>
   459→                  {PAYERS.map(p => (
   460→                    <SelectItem key={p} value={p}>{p}</SelectItem>
   461→                  ))}
   462→                </SelectContent>
   463→              </Select>
   464→            </div>
   465→          </div>
   466→
   467→          {/* Denial text */}
   468→          <div>
   469→            <label className="text-xs font-medium text-muted-foreground mb-1 block">Denial Letter</label>
   470→            <Textarea
   471→              value={denialText}
   472→              onChange={e => setDenialText(e.target.value)}
   473→              placeholder="Paste the denial letter text here..."
   474→              className="min-h-[120px] text-sm"
   475→            />
   476→          </div>
   477→
   478→          {/* Actions */}
   479→          <div className="flex flex-wrap gap-3">
   480→            <Button onClick={handleRun} disabled={loading || !denialText.trim()}>
   481→              {loading ? <><Loader2 className="h-4 w-4 mr-2 animate-spin" />Running...</> : 'Run Pipeline'}
   482→            </Button>
   483→            <Button
   484→              variant="outline"
   485→              onClick={handleGateTest}
   486→              disabled={gateTestLoading || !denialText.trim()}
   487→            >
   488→              {gateTestLoading ? <><Loader2 className="h-4 w-4 mr-2 animate-spin" />Testing...</> : <><Beaker className="h-4 w-4 mr-2" />Gate Test</>}
   489→            </Button>
   490→          </div>
   491→
   492→          {error && (
   493→            <div className="p-3 rounded-md bg-red-50 dark:bg-red-950/30 border border-red-200 dark:border-red-800 text-sm text-red-700 dark:text-red-300">
   494→              {error}
   495→            </div>
   496→          )}
   497→        </CardContent>
   498→      </Card>
   499→
   500→      {/* ── Agent Steps ──────────────────────────────────────── */}
   501→      {hasResult && (
   502→        <Card>
   503→          <CardHeader className="pb-3">
   504→            <CardTitle className="text-base">Pipeline Steps</CardTitle>
   505→          </CardHeader>
   506→          <CardContent>
   507→            <div className="space-y-1">
   508→              <AgentStep
   509→                name="Patient Advocate"
   510→                icon={<Heart className="h-4 w-4 text-white" />}
   511→                color="bg-rose-500"
   512→                status={hasResult ? 'completed' : 'idle'}
   513→                detail={result?.advocate.caseFraming.urgencyLevel ? `Urgency: ${result.advocate.caseFraming.urgencyLevel}` : undefined}
   514→              />
   515→              <AgentStep
   516→                name="Denial Triage"
   517→                icon={<Search className="h-4 w-4 text-white" />}
   518→                color="bg-teal-500"
   519→                status={hasResult ? 'completed' : 'idle'}
   520→                detail={result?.triage.denialJson.denialTypeLabel}
   521→              />
   522→              <AgentStep
   523→                name="HITL Gate 1"
   524→                icon={<Shield className="h-4 w-4 text-white" />}
   525→                color={gate1Status === 'approved' ? 'bg-emerald-500' : gate1Status === 'rejected' ? 'bg-red-500' : 'bg-amber-500'}
   526→                status={gate1Status === 'pending' ? 'pending' : gate1Status === 'approved' ? 'completed' : gate1Status === 'rejected' ? 'failed' : 'idle'}
   527→                detail={gate1Status === 'pending' ? 'Awaiting human confirmation' : gate1Status === 'approved' ? 'Approved' : gate1Status === 'rejected' ? 'Rejected' : undefined}
   528→              />
   529→              <AgentStep
   530→                name="Policy Research"
   531→                icon={<BookOpen className="h-4 w-4 text-white" />}
   532→                color="bg-emerald-500"
   533→                status={hasPolicy ? 'completed' : gate1Status === 'approved' ? 'running' : 'idle'}
   534→                detail={hasPolicy ? `${result!.policyResearch!.clauses.length} clauses retrieved` : undefined}
   535→                latencyMs={hasPolicy ? result!.policyResearch!.retrievalLatencyMs : undefined}
   536→              />
   537→              <AgentStep
   538→                name="Evidence Assembly"
   539→                icon={<Database className="h-4 w-4 text-white" />}
   540→                color="bg-cyan-500"
   541→                status={hasEvidence ? 'completed' : hasPolicy ? 'running' : 'idle'}
   542→                detail={hasEvidence ? `${result!.evidenceAssembly!.totalEvidenceItems} items, ${result!.evidenceAssembly!.evidenceStrength} strength` : undefined}
   543→              />
   544→              <AgentStep
   545→                name="Letter Drafting"
   546→                icon={<PenTool className="h-4 w-4 text-white" />}
   547→                color="bg-violet-500"
   548→                status={hasDraft ? 'completed' : hasEvidence ? 'running' : 'idle'}
   549→                detail={hasDraft ? `${result!.letterDrafting!.wordCount} words, ${result!.letterDrafting!.citationCount} citations` : undefined}
