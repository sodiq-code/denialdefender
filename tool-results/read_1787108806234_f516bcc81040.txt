   160→interface PermissionResult {
   161→  allowed: boolean;
   162→  reason: string;
   163→}
   164→
   165→function enforcePermission(agent: string, resource: Resource, capability: Capability): PermissionResult {
   166→  const scope = AGENT_SCOPES[agent];
   167→  if (!scope) return { allowed: false, reason: `Unknown agent: ${agent}` };
   168→  const caps = scope.resources[resource];
   169→  if (!caps || !caps.includes(capability)) {
   170→    return {
   171→      allowed: false,
   172→      reason: `DENIED: ${scope.role} does NOT have ${capability} permission on ${resource}. Action blocked at runtime.`,
   173→    };
   174→  }
   175→  return { allowed: true, reason: `${scope.role} has ${capability} permission on ${resource}` };
   176→}

   534→function mockWorkflow(input: Record<string, unknown>): Record<string, unknown> {

   543→    { step: 1, agent: "triage", timestamp: nowISO(), result_summary: { classification: triageResult.classification, confidence: triageResult.confidence, strategy: triageResult.strategy } },
   544→  ];

   564→
   565→  // Step 4: Evidence
   566→  const evidenceResult = mockEvidence({ denial, triage: triageResult });
   567→  decisionTraces.push({ step: 4, agent: "evidence", timestamp: nowISO(), result_summary: { evidence_count: evidenceResult.evidence_items.length, overall_strength: evidenceResult.overall_evidence_strength } });

   580→

   667→function corsHeaders(origin: string | null): Record<string, string> {
   668→  const allow = origin && (CORS_ORIGINS.includes(origin) || /.*\.run\.app$/.test(origin) || /.*\.preview\..*/.test(origin))
   669→    ? origin
   670→    : CORS_ORIGINS[0];
   671→  return {
   672→    "Access-Control-Allow-Origin": allow,
   673→    "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
   674→    "Access-Control-Allow-Headers": "Content-Type, Authorization",
   675→    "Access-Control-Allow-Credentials": "true",
   676→  };
   677→}
   678→
   679→function jsonResponse(data: unknown, status = 200, origin?: string | null): Response {
   680→  return new Response(JSON.stringify(data, null, 2), {
   681→    status,
   682→    headers: { "Content-Type": "application/json", ...corsHeaders(origin ?? null) },
   683→  });
   684→}

   686→async function parseBody<T = Record<string, unknown>>(req: Request): Promise<T> {
   687→  const text = await req.text();
   688→  return (text ? JSON.parse(text) : {}) as T;
   689→}

   691→// ─── Route handler ────────────────────────────────────────────────────────────
   692→async function handleRequest(req: Request): Promise<Response> {
   693→  const url = new URL(req.url);
   694→  const path = url.pathname;
   695→  const method = req.method;
   696→  const origin = req.headers.get("Origin");
   697→
   698→  // CORS preflight
   699→  if (method === "OPTIONS") {
   700→    return new Response(null, { status: 204, headers: corsHeaders(origin) });
   701→  }
   702→
   703→  try {
   704→    // GET /health
   705→    if (method === "GET" && path === "/health") {
   706→      return jsonResponse({
   707→        status: "ok",
   708→        service: SERVICE_NAME,
   709→        version: SERVICE_VERSION,
   710→        mock_mode: MOCK_MODE,
   711→        gemini_available: !MOCK_MODE,
   712→        model: GEMINI_MODEL,
   713→        port: PORT,
   714→        runtime: "bun",
   715→        permission_enforced: true,
   716→        permission_denials: permissionDenialLog.length,
   717→        agents: Object.keys(AGENT_FNS),
   718→        timestamp: nowISO(),
   719→      }, 200, origin);
   720→    }
   721→
   722→    // GET /permissions
   723→    if (method === "GET" && path === "/permissions") {
   724→      return jsonResponse({
   725→        permission_enforced: true,
   726→        scopes: AGENT_SCOPES,
   727→        primary_resources: AGENT_PRIMARY_RESOURCE,
   728→        recent_denials: permissionDenialLog.slice(-20),
   729→        total_denials: permissionDenialLog.length,
   730→        timestamp: nowISO(),
   731→      }, 200, origin);
   732→    }
   733→
   734→    // GET /gcp/status
   735→    if (method === "GET" && path === "/gcp/status") {
   736→      const status = await checkGcpStatus();
   737→      return jsonResponse(status, 200, origin);
   738→    }
   739→
   740→    // POST /agents/{name}
   741→    const agentMatch = path.match(/^\/agents\/([a-z]+)$/);
   742→    if (method === "POST" && agentMatch) {
   743→      const name = agentMatch[1];
   744→      const def = AGENT_FNS[name];
   745→      if (!def) {
   746→        return jsonResponse({
   747→          error: `Unknown agent: ${name}`,
   748→          available_agents: Object.keys(AGENT_FNS),
   749→        }, 404, origin);
   750→      }
   751→      const perm = enforcePermission(name, def.resource, def.capability);
   752→      if (!perm.allowed) {
   753→        permissionDenialLog.push({ agent: name, resource: def.resource, capability: def.capability, reason: perm.reason, timestamp: nowISO() });
   754→        return jsonResponse({ agent: name, status: "blocked", reason: perm.reason, permission_enforced: true }, 403, origin);
   755→      }
   756→      const body = await parseBody(req);
   757→      const start = Date.now();
   758→      const data = def.fn(body);
   759→      const latencyMs = Date.now() - start;
   760→      return jsonResponse({
   761→        agent: name,
   762→        status: "success",
   763→        data,
   764→        latencyMs,
   765→        permission_enforced: true,
   766→        trace: {
   767→          agent: name,
   768→          trace_id: uuid(),
   769→          elapsed_seconds: elapsedSince(start),
   770→          timestamp: nowISO(),
   771→          mode: "mock",
   772→          model: GEMINI_MODEL,
   773→        },
   774→      }, 200, origin);
   775→    }
   776→
   777→    // POST /workflow/run
   778→    if (method === "POST" && path === "/workflow/run") {
   779→      const body = await parseBody(req);
   780→      const start = Date.now();
   781→      const caseId = (body.case_id as string) ?? uuid();
   782→      const result = mockWorkflow(body);
   783→      const trace = (result._trace ?? {}) as Record<string, unknown>;
   784→      workflowStore.set(caseId, {
   785→        case_id: caseId,
   786→        workflow_id: (result.workflow_id as string) ?? uuid(),
   787→        status: (result.status as string) ?? "completed",
   788→        started_at: (trace.timestamp as string) ?? nowISO(),
   789→        updated_at: nowISO(),
   790→      });
   791→      const latencyMs = Date.now() - start;
   792→      return jsonResponse({
   793→        ...result,
   794→        latencyMs,
   795→        trace: { ...(result._trace as object), mode: "mock", model: GEMINI_MODEL, latencyMs },
   796→      }, 200, origin);
   797→    }
   798→
   799→    // GET /workflow/status/:id
   800→    if (method === "GET" && path.startsWith("/workflow/status/")) {
   801→      const id = path.replace("/workflow/status/", "");
   802→      const status = workflowStore.get(id);
   803→      if (status) return jsonResponse(status, 200, origin);
   804→      return jsonResponse({ error: `Workflow not found for id: ${id}` }, 404, origin);
   805→    }
   806→
   807→    // 404
   808→    return jsonResponse({
   809→      error: "Not found",
   810→      path,
   811→      available_endpoints: [
   812→        "GET  /health",
   813→        "GET  /permissions",
   814→        "GET  /gcp/status",
   815→        "POST /agents/{triage|coder|policy|evidence|citation|drafter|reviewer|orchestrator}",
   816→        "POST /workflow/run",
   817→        "GET  /workflow/status/:id",
   818→      ],
   819→    }, 404, origin);
   820→  } catch (err) {
   821→    console.error(`[server] Error handling ${method} ${path}:`, err);
   822→    return jsonResponse({ error: err instanceof Error ? err.message : "Internal server error" }, 500, origin);
   823→  }
   824→}
